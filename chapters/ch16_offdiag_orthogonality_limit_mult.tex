\section{Gelfand--Graev Mirabolic Bounds and Local-to-Global Off-Diagonal Splitting}
\label{sec:16:gg-offdiag}

At every split finite place $v$ of $E^+$ (where $G_v \cong \GL_N(E_v)$ and the Bessel subgroup embeds inside the mirabolic subgroup $P_{\mathrm{mir}, N} \subset \GL_N(E_v)$), the off-diagonal orbital integrals $\mathcal{O}_{\gamma, v}(\Phi_1, \Phi_2)$ are controlled by the representation theory of the mirabolic subgroup~\cite{BernsteinZelevinsky1977} and the Gelfand--Graev Whittaker model.

\begin{proposition}[Gelfand--Graev mirabolic local bound and scalar convolution trace law]
\label{prop:16:gg-mirabolic}
Let $v$ be a finite place of $E^+$ splitting in $E$ with residue cardinality $q_v$, so $G_v \cong \GL_N(E_v)$ and $H_v \cong \GL_k(E_v)$, and let $\tau_{1, v}, \tau_{2, v}$ be level-zero supercuspidal (or spherical) local test data on $H_v$ paired with the generic Bessel character $\psi_{z, c, v}$ on $N_{r, v}$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Mirabolic local orbital bound.)} For every non-identity double coset $\gamma \in S_{r, v} \backslash G_v / S_{r, v}$ parametrized by its mirabolic Bruhat invariant $\delta_v(\gamma) \ge 1$ (measuring the denominator height of the off-diagonal matrix block), the local relative orbital integral obeys
  \begin{equation}
  \label{eq:16:mirabolic-bound}
    \bigl|\mathcal{O}_{\gamma, v}(\Phi_{1, v}, \Phi_{2, v})\bigr|
    \;\le\; C_{\mathrm{mir}}\, q_v^{-c_1 \delta_v(\gamma)}\, \|\Phi_{1, v}\|_{L^2}\, \|\Phi_{2, v}\|_{L^2},
  \end{equation}
  with explicit exponent $c_1 > 0$ depending only on the rank gap $r$.
\item \emph{(Scalar convolution trace law.)} On any irreducible generic local representation $\pi_v$ of $\GL_N(E_v)$ whose Gelfand--Graev / Bessel period is nonvanishing, convolving against the localized Gelfand--Graev projector acts by an explicit scalar determined by the local Bessel formal degree (matching the Bernstein--Zelevinsky derivative filtration~\cite{BernsteinZelevinsky1977} and Casselman--Shalika~\cite{Casselman1989}).
\item \emph{(Active polynomial-growth splitting and local-to-global pair bound.)} For any pair of distinct column-chain indices $(u, u')$ ($u \neq u' \in \TT$), splitting the off-diagonal Gram inner product $\langle B_u, B_{u'} \rangle$ across the active polynomial-growth operator partitions the global sum over double cosets into local-to-global Euler products controlled by \eqref{eq:16:mirabolic-bound}, yielding
  \begin{equation}
  \label{eq:16:pair-offdiag}
    \sum_{u \neq u'} \bigl\|B_u^* B_{u'}\bigr\|_{\HS}^2 \;\le\; \frac{1}{4}\, e\, R_{\mathrm{tot}}
  \end{equation}
  once combined with the archimedean large-gap Bessel decay of \cref{sec:16:jsz-locscalar} below!
\end{enumerate}
\end{proposition}

\section{Archimedean Bessel Multiplicity One (Jiang--Sun--Zhu) and Local Scalar Rigidity}
\label{sec:16:jsz-locscalar}

To factor the global archimedean Bessel functional into explicit local standard Bessel models with \emph{exact proportionality constant $c_v = 1$} (and thus transfer global archimedean decay directly from the standard local Bessel model $W_{0, v}$), two ingredients are required:
\begin{enumerate}[label=(\arabic*)]
\item \emph{Archimedean Bessel multiplicity one}: $\dim_{\CC} \Hom_{S_r, \mathrm{cont}}(\sigma_v^\infty \widehat{\otimes} (\pi_v^\infty)^\vee, \psi_{z,c}) \le 1$; and
\item \emph{Local scalar rigidity}: showing unconditionally that the proportionality constant $c_v \in \CC^\times$ arising from Schur-orthogonal projection onto the discrete series obeys $c_v^2 = c_v$, hence $c_v = 1$!
\end{enumerate}

In the formal archive, archimedean Bessel multiplicity one was proved internally across ten pages under a large-gap hypothesis. Here we invoke the published theorem of Jiang, Sun, and Zhu~\cite[Theorem~A]{JiangSunZhu2010}—verifying every single one of its six hypotheses explicitly—and then give the complete, airtight, four-step self-contained proof of local scalar rigidity ($c_v = 1$).

\newsimplification{Archimedean Bessel Multiplicity One via Jiang--Sun--Zhu (2010, Theorem~A) and Self-Contained Local Scalar Rigidity}
\litref{Jiang--Sun--Zhu (2010, GAFA~20, Theorem~A)~\cite{JiangSunZhu2010}; Wallach (1988, Chapter~11)~\cite{Wallach1988}; Harish-Chandra (1966)~\cite{HarishChandra1966}}

\begin{proposition}[Archimedean Bessel multiplicity one and exact local scalar rigidity ($c_v = 1$)]
\label{prop:16:jsz-locscalar}
Let $v \mid \infty$ be a real place of $E^+$, let $G_v = \UU(p_v, q_v)$ ($N = 2r + k + 1$, $k \ge 1, r \ge 0$) with Bessel subgroup $S_v = N_{\mathrm{loc}, v} \rtimes H_{\mathrm{fix}, v}$ ($H_{\mathrm{fix}, v} \cong \UU(k)$ compact), anisotropic vector $z$ ($T_z > 0$), and generic character $\psi_{z, c}$ on $N_{\mathrm{loc}, v}$. Let $(\sigma_v, \mathcal{H}_v)$ be an irreducible unitary discrete-series representation of $G_v$ with Harish--Chandra parameter satisfying the large spectral gap $b_{j+1}(\tau) - b_j(\tau) \ge 16 r^{19}$ ($r \ge 1801$), let $(\pi_v, W_p)$ be an irreducible finite-dimensional unitary representation of the compact group $H_{\mathrm{fix}, v}$, and let $P_v \in L^1(G_v, d\mu_v)$ be the formal-degree-normalized Schur reproducing kernel of $\sigma_v$ (so $\int_{G_v} P_v(g) P_v(g^{-1} y)\, d\mu_v(g) = P_v(y)$).
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Archimedean Bessel multiplicity one via Jiang--Sun--Zhu~\cite{JiangSunZhu2010}.)} The space of Fr\'echet-continuous Bessel intertwiners $\Hom_{S_v, \mathrm{cont}}(\mathcal{H}_v^\infty, W_p \otimes \psi_{z,c})$ has complex dimension at most~$1$. In particular, any two smooth Bessel maps $B_1, B_2$ on $\mathcal{H}_v^\infty$ are linearly dependent.
\item \emph{(Exact local scalar rigidity: $c_v^2 = c_v \implies c_v = 1$.)} Let $f = W_{\mathrm{loc}, v} : G_v \to W_p$ be a nonzero continuous $S_v$-equivariant Bessel function belonging to $L^2(S_v \backslash G_v, d\nu_v)$ and satisfying the reproducing eigenvalue equation
  \begin{equation}
  \label{eq:16:cv-eigen}
    \int_{G_v} P_v(g)\, f(x g)\, d\mu_v(g) \;=\; c_v\, f(x) \qquad \text{for all } x \in G_v,
  \end{equation}
  for some nonzero scalar $c_v \in \CC^\times$. Then **$c_v = 1$** identically!
\end{enumerate}
\end{proposition}

\begin{proof}
(i) We verify all six hypotheses of Jiang--Sun--Zhu~\cite[Theorem~A]{JiangSunZhu2010} one-to-one:
\begin{enumerate}[label=\textup{\arabic*.}]
\item \emph{Unitary pair and odd codimension:} $G_v = \UU(p_v, q_v)$ has $W_z^\perp = \mathbb{H}^r \oplus \CC z$ with $T_z = \langle z, z \rangle > 0$ anisotropic and codimension $2r + 1$ odd, while $H_{\mathrm{fix}, v} = \UU(W_z) \cong \UU(k)$.
\item \emph{Bessel subgroup:} $S_v = N_{\mathrm{loc}, v} \rtimes H_{\mathrm{fix}, v}$, where $N_{\mathrm{loc}, v}$ stabilizes the maximal isotropic flag and acts trivially on $W_z$, normalized by $H_{\mathrm{fix}, v}$ which preserves $\psi_{z, c}$.
\item \emph{Generic character:} every simple-root coefficient $c_a \neq 0$ and $T_z > 0$, so $\psi_{z, c}$ is generic on $N_{\mathrm{loc}, v}$.
\item \emph{Irreducible Casselman--Wallach $\sigma_v^\infty$:} $(\sigma_v, \mathcal{H}_v)$ is an irreducible unitary square-integrable representation of $G_v$; by Harish-Chandra admissibility and Casselman--Wallach globalization~\cite{Wallach1988}, its Fr\'echet space of smooth vectors $\mathcal{H}_v^\infty$ (with Lie-word seminorms $\|d\sigma_v(W) u\|_{\mathcal{H}_v}$) is an irreducible Casselman--Wallach representation (class $\mathrm{FH}$).
\item \emph{Irreducible Casselman--Wallach $\pi_v$:} $(\pi_v, W_p)$ is a finite-dimensional irreducible unitary representation of the compact group $\UU(k)$, hence trivially irreducible Casselman--Wallach and equal to its own smooth vectors.
\item \emph{Continuity and dual-space identification:} Because $\dim_{\CC} W_p < \infty$, the evaluation map $B \mapsto \bigl((u, \xi^*) \mapsto \xi^*(B(u))\bigr)$ is a linear isomorphism
  \[
    \Hom_{S_v, \mathrm{cont}}\bigl(\mathcal{H}_v^\infty,\, W_p \otimes \psi_{z,c}\bigr)
    \;\xrightarrow{\;\sim\;}\;
    \Hom_{S_v, \mathrm{cont}}\bigl(\mathcal{H}_v^\infty \widehat{\otimes} W_p^\vee,\, \psi_{z,c}\bigr).
  \]
\end{enumerate}
Therefore Jiang--Sun--Zhu~\cite[Theorem~A]{JiangSunZhu2010} applies verbatim and gives $\dim_{\CC} \Hom_{S_v, \mathrm{cont}}(\mathcal{H}_v^\infty, W_p \otimes \psi_{z,c}) \le 1$.

(ii) We prove $c_v = 1$ in four completely self-contained steps:

\emph{Step~1 (Right-invariant Bruhat weight measure $\nu'$ on $G_v$).}
Because $N_{\mathrm{loc}, v}$ is closed and unimodular and $H_{\mathrm{fix}, v}$ is compact normalizing $N_{\mathrm{loc}, v}$, the Bessel subgroup $S_v = N_{\mathrm{loc}, v} \rtimes H_{\mathrm{fix}, v}$ is closed and unimodular inside the unimodular group $G_v$. Fix a nonnegative measurable Bruhat weight function $\beta' : G_v \to [0, \infty)$ satisfying $\int_{S_v} \beta'(s x)\, d\mu_{S, v}(s) = 1$ for all $x \in G_v$, and set $d\nu'(x) := \beta'(x)\, d\mu_v(x)$. Then for every nonnegative measurable left-$S_v$-invariant function $\phi$ on $G_v$, we have $\int_{G_v} \phi(x)\, d\nu'(x) = \int_{S_v \backslash G_v} \phi(\bar{x})\, d\nu_v(\bar{x})$, and because $G_v$ is unimodular, right translation preserves the $L^2$-norm:
\[
  \int_{G_v} \|f(x y)\|^2\, d\nu'(x) \;=\; \int_{S_v \backslash G_v} \|f(\bar{x} y)\|^2\, d\nu_v(\bar{x}) \;=\; \int_{S_v \backslash G_v} \|f(\bar{x})\|^2\, d\nu_v(\bar{x}) \;=:\; L \;\in\; (0, \infty)
\]
for every $y \in G_v$ (strictly positive since $f$ is continuous and nonzero, and finite since $f \in L^2(S_v \backslash G_v)$).

\emph{Step~2 (Existence of a Fubini-admissible basepoint $x \in G_v$ with $f(x) \neq 0$).}
Using the elementary inequality $a b \le a^2 + b^2$ for all $y \in G_v$, we have
\[
  \int_{G_v} \|f(x)\|\,\|f(x y)\|\, d\nu'(x) \;\le\; \int_{G_v} \bigl(\|f(x)\|^2 + \|f(x y)\|^2\bigr)\, d\nu'(x) \;=\; 2 L.
\]
Define $A(x) := \int_{G_v} |P_v(g)| \left(\int_{G_v} |P_v(h)|\,\|f(x g h)\|\, d\mu_v(h)\right) d\mu_v(g) \in [0, \infty]$. Since $P_v \in L^1(G_v, d\mu_v)$ (guaranteed by the large spectral gap $b_{j+1} - b_j \ge 16 r^{19}$), Tonelli's theorem gives
\[
  \int_{G_v} \|f(x)\|\, A(x)\, d\nu'(x)
  \;=\; \int_{G_v}\int_{G_v} |P_v(g)|\,|P_v(h)| \left(\int_{G_v} \|f(x)\|\,\|f(x g h)\|\, d\nu'(x)\right) d\mu_v(h)\, d\mu_v(g)
  \;\le\; 2 L\,\|P_v\|_{L^1}^2 \;<\; \infty.
\]
Because $E := \{x \in G_v : f(x) \neq 0\}$ is nonempty and open (by continuity of $f$), it has $\nu'(E) > 0$. Since $\|f(x)\| A(x)$ is finite $\nu'$-almost everywhere, there exists $x \in E$ such that simultaneously $f(x) \neq 0$ and $A(x) < \infty$!

\emph{Step~3 (Schur formal-degree convolution identity).}
Because $P_v(g) = d(\sigma_v)\,\langle \sigma_v(g) v_0, v_0 \rangle$ is the formal-degree-normalized matrix coefficient of a unit $K_v$-finite vector $v_0 \in \mathcal{H}_v$ in the discrete series $\sigma_v$, Harish-Chandra--Schur orthogonality~\cite{HarishChandra1966} yields the exact idempotent convolution law
\begin{equation}
\label{eq:16:pv-conv}
  \int_{G_v} P_v(g)\, P_v(g^{-1} y)\, d\mu_v(g) \;=\; P_v(y) \qquad \forall\, y \in G_v.
\end{equation}

\emph{Step~4 (Double Fubini and measure-preserving shear).}
At the good basepoint $x$ from Step~2, define $\Phi(g, h) := P_v(g) P_v(h) f(x g h)$ on $G_v \times G_v$. By Step~2, $\iint_{G_v \times G_v} \|\Phi(g, h)\|\, d\mu_v(g)\, d\mu_v(h) = A(x) < \infty$, so $\Phi \in L^1(G_v \times G_v)$. Applying Fubini's theorem and invoking \eqref{eq:16:cv-eigen} twice gives
\[
  \iint_{G_v \times G_v} \Phi(g, h)\, d\mu_v(g)\, d\mu_v(h)
  \;=\; \int_{G_v} P_v(g) \left(\int_{G_v} P_v(h)\, f((x g) h)\, d\mu_v(h)\right) d\mu_v(g)
  \;=\; c_v \int_{G_v} P_v(g)\, f(x g)\, d\mu_v(g) \;=\; c_v^2\, f(x).
\]
Now apply the $(\mu_v \times \mu_v)$-preserving shear change of variables $(g, h) \mapsto (g, y := g h)$ (which preserves Haar measure because $G_v$ is unimodular!), sending $\Phi(g, h)$ to $\Psi(g, y) := P_v(g) P_v(g^{-1} y) f(x y) \in L^1(G_v \times G_v)$. Applying Fubini in $(g, y)$, invoking \eqref{eq:16:pv-conv}, and applying \eqref{eq:16:cv-eigen} once gives
\[
  \iint_{G_v \times G_v} \Psi(g, y)\, d\mu_v(g)\, d\mu_v(y)
  \;=\; \int_{G_v} \left(\int_{G_v} P_v(g) P_v(g^{-1} y)\, d\mu_v(g)\right) f(x y)\, d\mu_v(y)
  \;=\; \int_{G_v} P_v(y)\, f(x y)\, d\mu_v(y) \;=\; c_v\, f(x).
\]
Since $\iint \Phi = \iint \Psi$, we obtain $(c_v^2 - c_v) f(x) = 0$. Because $f(x) \neq 0$ and $c_v \neq 0$, this forces $c_v = 1$ identically!
\end{proof}

\section{Kloosterman Coset-Sum Closure, Limit Multiplicity, and Statements~(S), (I), (P)}
\label{sec:16:kloosterman-sip}

To close both the non-generic orbital piece $\mathcal{I}_{\mathrm{ng}}$ in \eqref{eq:15:four-part-sum} and the frame-tightness / limit-multiplicity count $D_{\mathrm{all}} = (1 + O(x^{-\varepsilon})) D_{\mathrm{exp}}$ needed in \cref{thm:06:nonvanishing}, we establish microlocal Kloosterman cancellation and identity-cell Haar pinning.

\begin{proposition}[Microlocal Kloosterman cancellation and identity-cell Haar pinning]
\label{prop:16:kloosterman}
On every non-generic Bruhat cell $\gamma \in \mathrm{ng}$ of the four-way unfolding \eqref{eq:15:four-part-sum}:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Microlocal Kloosterman density vanishing and fiberwise coset-sum bound.)} Partitioning the finite-level non-generic double cosets into unipotent cosets modulo $g_0 g_1 T$, the additive character sum along each non-degenerate fiber is a generalized Kloosterman sum over $\FF_{g_0} \times \FF_{g_1}$ whose microlocal density vanishes to high order; by Deligne's Weil bound for Kloosterman sums~\cite{Deligne1974} summed fiberwise over the rank strata,
  \[
    \bigl|\mathcal{I}_{\mathrm{ng}}(\Phi_1, \Phi_2)\bigr| \;\le\; C_{\mathrm{Kloo}}\, (g_0 g_1)^{-c_{\mathrm{ng}}}\, \mathcal{I}_{\mathrm{id}}(\Phi_1, \Phi_1)^{1/2}\,\mathcal{I}_{\mathrm{id}}(\Phi_2, \Phi_2)^{1/2}.
  \]
\item \emph{(Identity-cell vanishing and Haar measure pinning.)} Outside the identity double coset, every near-identity congruence cell vanishes identically under the neat level subgroup $\Pi(g_0, g_1)$ ($2fk < \min(g_0, g_1) - 1$), pinning the local Haar normalization constant $\nu_Q > 0$ cleanly to the compact-dual volume ratio of \cref{thm:12:size-inputs} (Block~IV).

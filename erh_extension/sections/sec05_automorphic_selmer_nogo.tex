\subsection{Real-Quadratic Count~I (\texttt{BK-HMF-Odd}) Across All Bases $K_0 = \QQ(\sqrt{D})$}\label{sec:5:count1}

In the ported architecture, Count~I (\texttt{BK-HMF-Odd}) requires $H^1_f(F^+, W) = H^1_f(F^+, W \otimes \theta) = 0$ for an odd-weight parallel Hilbert modular form $W$ of weight $(2e+1, \dots, 2e+1)$ induced along a solvable base-change tower $L^+/F^+$ (where $\ell \ge 5$, $(\ell-1)\nmid 2e$, and $\theta$ is the CM quadratic character).

\begin{theorem}[Real-Quadratic Count~I Master Assembly and Characteristic-$0$ Boundary~\texttt{E01--E03, F02, F06--F07, H04, H07}]\label{thm:5:rq-count1}
\tagProved{} \textup{(Split 50\%)} / \tagMod{} \textup{(Inert 50\% modulo BP~\cite{BP2021} \& GR~\cite{GR2024} and char-0 dichotomy)}
Let $K_0 = \QQ(\sqrt{D})$ be \emph{any} real quadratic field ($D > 1$ squarefree, arbitrary narrow class number $h^+(K_0) \ge 1$, allowing $2 \mid D_{K_0}$ and $K_0 = \QQ(\sqrt{2})$), and let $\ell \ge 5$ with $(\ell-1)\nmid 2e$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Master Scheme~I over $\GSp_4/\QQ$.)} Perturbing $W$ in a $2$-parameter Hida family over $K_0$ to non-parallel regular weight $(k_1, k_2) = (2e+1+\ell^M, 2e+1-\ell^M)$ and applying automorphic induction $\Pi = \AI_{K_0/\QQ}(\pi_f)$ to $\GSp_4/\QQ$ yields Bloch--Kato vanishing $H^1_f(K_0, W) = H^1_f(K_0, W\otimes\theta) = 0$ (via multiplicity-free Morita quotient projection $\lambda$ stripping spurious congruence factors~\cite{D02} and solvable base change $L^+/K_0$) up to the characteristic-$0$ interface: either \textup{(A)} automorphy $(\mathrm{Aut}_3)_{L^+}$ for symmetric squares / base change, or \textup{(B)} Greenberg--Mazur cyclotomic $\Lambda_{\cyc}$-torsionness of $X_{\Gr}(W^*(1))$ at parallel weight. The split-$50\%$ prime branch ($\ell$ split in $K_0$, using Loeffler--Zerbes~\cite{LZ2026} and Rockwood) uses \emph{zero preprints}; the inert-$50\%$ branch ($\ell$ inert in $K_0$, Spin $p$-adic $L$-function in Region~(F)) relies on Boxer--Pilloni~\cite{BP2021} and Graham--Rockwood~\cite{GR2024} (both still preprints as of Nov~2026). This completely removes the $h^+(K_0)=1$ and $\gcd(D_{K_0}, |D_{E_0}|)=1$ restrictions.
\item \emph{(Tame Hecke invariance and LDS four-fold obstruction \tagProved{}.)} In the Morita-projected Euler system $\mathbf{c}_\lambda(T) = \mathbf{c}_{\mathrm{geom}}(T)\mathbf{c}_{\mathrm{ES}}(T)\mathbf{c}_{\mathrm{per}}(T) = T^m u(T)$, the defect $\delta_{\mathrm{def}} := m - m_{\mathrm{glob}}$ is strictly invariant under all tame Hecke operators $h \in A_D$ (cross-branches share the same Hecke eigenvalues at $T=0$). Moreover, $m_{\mathrm{ES}} \ge 1$ is purely local at $\ell$, so $m_{\mathrm{glob}} = m$ cannot be deduced formally without Greenberg--Mazur torsionness. At the limit discrete series (LDS) weight $(2e+1, 2)$ ($r_2 = -1$), four obstructions block \texttt{Lemma-ES-LDS}: Asai non-criticality, Serre-duality sheaf collision jumping $\dim H^3_{\mathrm{cl}}[\Pi_0]$ from $0$ to $1$ and violating strict small-slope $0 < 0$, absence of classical local systems for $H^3_{\et}$, and double $p$-stabilization roots $\tilde\alpha_1 = \tilde\alpha_2$ at $50\%$ of split primes under twisted base change. Finally, published $\GL_4$ Shalika families~\cite{BDGJW2025,BDW2026} are restricted to tame level $1$, whereas real quadratic conductors satisfy $\ge D^2 \ge 25$.
\end{enumerate}
\end{theorem}

\subsection{Higher-Degree ($d_0 \ge 3$) Definite-Quaternion $\mu_{\an}^{\mathrm{opt}} = 0$ and Obstructions}\label{sec:5:higher}

For $d_0 = [F^+:\QQ] \ge 3$, Route~4A of~\texttt{F04} fixes a parity mismatch in earlier arguments via a prime-to-$\ell$ cyclotomic base change $L' = L(\zeta_\ell)$, $L'^+ = (L')^{\langle c \rangle}$ (of degree $d_{L'^+} = (\ell-1)d_L \equiv 0 \pmod 4$, with $L'/L'^+$ unramified at all finite places and all $v' \mid \ell$ split in $L'/L'^+$), transporting $W$ to a parallel weight-$2$, trivial-central-character, root-number-$+1$, residually reducible cusp form $F_2 = \BC_{L'^+/L^+}(f_2)$ on $L'^+$. On the analytic side, we overcome the Eisenstein constant-term obstruction completely:

\begin{theorem}[Definite-Quaternion Eisenstein Spectral Dichotomy and $\mu_{\an}^{\mathrm{opt}} = 0$~\texttt{H08-1}]\label{thm:5:quat-mu}
\tagProved{} \textup{(Full ring-class algebra and $s=1$)} / \tagMod{} \textup{($s > 1$ orbit non-constancy)}
Since $4 \mid d_{L'^+}$, there exists a unique definite quaternion algebra $B/L'^+$ ramified at all $\infty$ places and split at all finite places, carrying a primitive integral Hecke eigenform $\phi_{F_2} \in \mathrm{Map}(X_U, \OO)$ with $\bar\rho_{F_2}^{\mathrm{ss}}|_{G_{L'^+}} \cong (\theta' \bar\psi) \oplus \bar\psi^{-1}$ ($\theta' = \omega_\ell|_{G_{L'^+}}$ quadratic).
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Spectral dichotomy and inertia-vanishing normalization.)} On the Eisenstein subspace $\mathcal{E}(X_U, k) = \mathrm{Map}(\Cl^+(L'^+), k)$, $T_{\QQp}|_{\mathcal{E}} = (1 + \Nm\QQp)[\QQp] \equiv (1 + \theta'(\QQp))[\QQp] \equiv 0 \pmod{\vpi}$ at all inert $\QQp \nmid \ell$. Hence if $\bar\psi^2|_{G_{L'^+}} \neq \mathbf{1}$ (Region~I), $\phi_{F_2} \bmod \vpi \notin \mathcal{E}(X_U, k)$ automatically ($m_{\Eis} = 0$). In general (Region~II), subtracting the principal-coset Eisenstein constant $\phi_{\Eis} \in \mathcal{E}(X_U, \OO)$ and dividing by $\vpi^{m_{\Eis}^{\mathrm{prin}}}$ ($< \infty$ since $F_2$ is non-CM) yields $\widetilde{\phi}_{F_2}$ satisfying exact zero-error inertia cancellation $(\tau-1)\widetilde{\Theta}_{F_2, n} = (\tau-1)\Theta_n(\widetilde{\phi}_{F_2})$ ($\forall \tau \in K_n$) and strict non-constancy of $\widetilde{\phi}_{F_2}^{(\ell)} \bmod \vpi$ on every principal-coset Bruhat--Tits fiber (with zero residual irreducibility needed!).
\item \emph{(Analytic $\mu_{\an}(\widetilde{\Theta}_{F_2}) = 0$ and full-tower propagation.)} By Cornut--Vatsal~\cite{CV2007}, $\widetilde{\Theta}_{F_2} \not\equiv 0 \pmod{\vpi}$ inside $\OO[[\mathcal{G}_{\ac, v'_0}]]$ for all $d_0 \ge 1$. On $\OO[[\Gamma_{\ac, v'_0}]]$, $\mu_{\Lambda_{\ac, v'_0}}(\widetilde{\Theta}_{F_2, \mathbf{1}}) = 0$ holds unconditionally when $s = |\Delta_{\mathrm{fin}} \cap \langle [w'_0]\rangle| = 1$, and holds for $s > 1$ modulo single $D_{w'_0}$-orbit-sum non-constancy. Moreover, projecting $\Gamma_{\ac}(L') \twoheadrightarrow \Gamma_{\ac}(L)$ can send $\mu$ to $\infty$ (via $\Delta$-invariant polynomials vanishing on the diagonal), so \texttt{F04} Theorem~F04-1B operates on the full tower $\Lambda_{\ac}(L')$, propagating $\mu_{\alg}(X_{\Gr}(L'\cdot L_{\ac}, T_{\mathrm{opt}})) = 0$ back to \texttt{BK-HMF-Odd} modulo $(\mathrm{Aut}_3)_{L'^+}$.
\end{enumerate}
\end{theorem}

Despite Theorem~\ref{thm:5:quat-mu} on the analytic side, the algebraic and automorphic bridges in degree $d_0 \ge 3$ hit rigid structural walls:

\begin{theorem}[Higher-Degree ($d_0 \ge 3$) Algebraic and Automorphic Obstructions~\texttt{F04-3B, H08-2/3, N03-B}]\label{thm:5:d3obs}
\tagProved{} \textup{(Obstruction theorems)} / \tagObs{} \textup{(Routes)}
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{($\zeta_\ell$-Admissible Prime Catch-22 \texttt{H08-2A}.)} Crossing the shift $\Delta s = 1/2$ modulo $\vpi$ forces $\zeta_\ell \in L'$ (\texttt{F04-1A}). At \emph{every} prime $\QQp \nmid \ell$ inert in $L'/L'^+$, $\Frob_{\QQp}$ acts as $c = -1$ on $\mu_\ell$, forcing $\Nm\QQp \equiv -1 \pmod{\ell}$ and $\Nm\mathfrak{Q} = (\Nm\QQp)^2 \equiv +1 \pmod{\ell}$. Consequently, the Bertolini--Darmon / Howard / LTXZZ admissible prime condition $\ell \nmid (\Nm\QQp^2 - 1)$ holds for \emph{zero} primes (the admissible set is $\emptyset$), finite--ordinary transversality collapses ($\dim_k(H^1_{\mathrm{fin}} \cap H^1_{\mathrm{ord}}) = 1 \neq 0$), and $H^1_{\mathrm{fin}}(L'_{\mathfrak{Q}}, \OO(1)/\vpi^n) \cong \OO/(\ell)$ has nonvanishing kernel of length $\ge e_\OO \ge 1$ at every layer.
\item \emph{(Burnside--$\mu$ incompatibility and bipartite collapse \texttt{H08-2B/2C}.)} For any residually reducible lattice $T$, $\OO[H] \bmod \vpi \subseteq \mathfrak{b}_2(k) \subsetneq M_2(k)$ forces Burnside index $c_{\End}(T) \ge 1$, so characteristic-$0$ Kolyvagin/Rubin arguments incur an additive error $\ge 1$ that vanishes for $\lambda$ over $\Lambda_{\ac}[1/\ell]$ but never yields $\mu_{\alg} = 0$. Furthermore, CGLS~\cite{CGLS2022} and CGS~\cite{CGS2025} use Heegner + Rubin~\cite{Rubin1991} elliptic units + Beilinson--Flach (not Kato), all of which fail on $F_2/L'^+$ ($d_{L'^+} \ge 4$).
\item \emph{(Bott-periodicity non-transfer to $\GSpin(5,2)$ \texttt{F04-3B, H08-3}.)} For $d_0 \ge 3$, $\AI_{F_3^+/\QQ}(W)$ lifts to a symplectic $6$-dimensional representation into $\GSp_6 = \widehat{\GSpin(5,2)}^\circ$ (for both cyclic $A_4$ and non-Galois $S_3$ cubics via Mackey), yet the real Hasse--Witt invariant obeys the period-$4$ Bott law
\[
  c_\infty\bigl(\SO(2d_0-1, 2)\bigr) \;=\; (-1)^{\lfloor (d_0-1)/2 \rfloor}, \qquad\text{so } c_\infty\bigl(\SO(5,2)\bigr) = -1.

Hence every $\QQ$-form of $\SO(5,2)$ is non-split at an odd number ($\ge 1$) of finite primes $p_0$, rendering the Siegel Levi $\GL_3 \times \GL_1$ irrelevant, whereas \texttt{ch29} Count~I forms $W$ are unramified principal series everywhere and cannot transfer to any $\QQ$-form of $\GSpin(5,2)$. \subsection{The Four-Pillar Reductive Group Architecture No-Go Theorem}\label{sec:5:nogo}

A natural question for Wall~B (\texttt{I2}) and \texttt{I3a} in Regime~III/IV is whether replacing diagonal $\mathcal{G}_3$ (Count~V) or diagonal $\GSp_4$ (Count~II) in \texttt{ch29}--\texttt{ch30} by an off-diagonal block decomposition or a higher-rank classical reductive group $\GL_N, \mathcal{G}_N, \GSp_{2n}, \GO_N$ can bypass $\mathrm{EffWash}_{p_0,\ell}(K_0, \eta)$. The answer is unconditionally negative:

\begin{theorem}[Four-Pillar Reductive Group Architecture No-Go Theorem~\texttt{H06-3}]\label{thm:5:4pillar}
\tagProved{}
Across all classical reductive groups $G \in \{\GL_N, \mathcal{G}_N, \GSp_{2n}, \GO_N\}$ and all off-diagonal or higher-rank block partitions ($2+2$, $3+1$, $2+1+1$, $2+2+2$, $3+3$, $2+2+2+2$, etc.), diagonal $\mathcal{G}_3$ (Count~V) and diagonal $\GSp_4$ (Count~II) form the unique minimal viable architecture:
\begin{enumerate}[label=\textbf{Pillar~\arabic*.}]
\item \emph{(Unpolarized $\GL_N$ defect, $N \ge 3$.)} Every non-polarized $\GL_N$ has Calegari--Geraghty defect $\ell_0(N) = \dim \mathfrak{sl}_N^{c_v = +1} - \dim \mathfrak{u}^+ \ge \lfloor N/2 \rfloor - 1 \ge 1$, giving total Euler defect $\Delta_{\tot} = \dim Z(M) - d_0 \ell_0(N) < 0$ (e.g.\ $-1$ at $N=4, d_0=2$), breaking regular algebraic automorphy and Lemma~M.
\item \emph{(Principal polarization sign-parity law.)} In any polarized group $G \in \{\mathcal{G}_{2m}, \GSp_{2m}, \GO_{2m}\}$, if $\bar\Psi^{-1}$ embeds between dual $m$-dimensional regular algebraic blocks $(W_j, W_i)$ in $\Hom(W_j, W_i)\cap\mathfrak{g}_{-1}$, Euler--Poincar\'e balance enforces $\tau_Q \mu(c_v) = +1$, whence complex conjugation $c_v$ acts on the scalar line $\phi_0 \in \Hom(W_m, \bar\Psi^{-1}W_m)$ by
\[
  \iota_B(\phi_0) \;=\; -\sigma \cdot \tau_Q \cdot \mu(c_v) \cdot \phi_0 \;=\; -\phi_0,
\]
forcing the character to be totally \emph{odd} ($\bar\Psi^{-1}\eta_K$) rather than totally even ($\bar\Psi^{-1}$), blocking all paired $m+m$ and Klingen partitions.
\item \emph{(Uniqueness $d_+(\mathfrak{g}_{-2}) = 0 \iff m = 1$ and Lemma~M collapse.)} The graded defect dimension in degree $-2$ equals $d_+(\mathfrak{g}_{-2}) = \frac{m(m-1)}{2}$ (for $\mathcal{G}_N$) and $\lfloor m/2 \rfloor \lceil m/2 \rceil$ (for $\GSp_{2n}$). Hence $m = 1$ is the \emph{unique} block dimension with $d_+(\mathfrak{g}_{-2}) = 0$. For all $m \ge 2$, $\Delta_{\mathrm{ord}}(\mathfrak{g}_{-2}) = -e_{-2} \le -d_0 < 0$ injects factors $(1-x^2)^{e_{-2}}$ into the multi-Rees generating series $P(x,y)$, cancelling the $(1-x)^{-1}$ pole so $P(x,y) \in \ZZ[x][[y]]$ is polynomial in $x$ and the Koszul cotangent lower bound vanishes identically (explicitly witnessed by $I = (\vpi^N y_0, y_0^2 z_1, y_0^2 z_2, y_0^2)$ with $\mathrm{cotLen} = N \to \infty$). Moreover, weight $-1$ forces adjoint blocks $\bar\Psi^{-1} \otimes \ad^0(\bar\rho_g)$ requiring open non-abelian twisted $\mathrm{Sym}^2(g)\otimes\psi^{-1}$ Bloch--Kato vanishing.
\item \emph{(Higher-rank diagonal $\mathcal{G}_4$ and $\GSp_{2n}$ quotient locks.)} In diagonal $\GSp_{2n}$ ($n \ge 2$), the two odd roots satisfy $R_a/R_b = \bar\Psi$, forcing at least one branch in $\widehat{G_{K_0}}/(\widehat{G_\QQ}\circ N)$ to carry $[\psi_{\mathrm{fix}}] \neq 0$ (returning to $\mathrm{EffWash}$). In diagonal $\mathcal{G}_4$ ($1+1+1+1$), exhaustive search over all $440$ parameters $(a,b)$ shows zero $1$-parameter subtori satisfy \texttt{(M2)}, \texttt{(M3)}, and $\Delta_{<-1}=0$, while a $2$-torus Type~IIa block $W_2 \oplus (W_2^c)^\vee$ is non-polarized over CM $K$ with $\ell_0 = d_0 = 2 > 0$.
\end{enumerate}
\end{theorem}

\chapter{Coarse Multiplicity, Kernel-Free Gain, and the Glued Lattice Theorem}
\label{ch:09}

In \cref{ch:08} we extracted exact inert-prime derivative divisibility along individual cyclic Frobenius orbits. Two structural challenges remain before those local divisibility bounds can be fed into the lower face \eqref{eq:06:lower-face} of the Two-Sided Arakelov Slope Sandwich:
\begin{enumerate}[label=(\roman*)]
\item \emph{Coarse multiplicity and kernel-free decoupling.} The evaluation maps $B_u$ carry large polynomial multiplicities across all $m$ layers of the tower, and a priori the local column span could lose rank on a small exceptional locus of primes or degenerate orbits. We must prove that the bad window share is bounded by $\le x^{-9}$ (\cref{thm:09:window-share})—closing all four technical sub-engines in completely self-contained fashion—and then decouple the $m$-step gain from the explicit theta kernels via the \emph{zero-chain decoupling principle} (\cref{thm:09:kernel-free-gain}).
\item \emph{Global gluing and $D_\Gamma$ cancellation.} Local divisibility holds at each finite place $v$ individually, with a global denominator $D_\Gamma$ coming from the Galois-orbit normalization. We must glue all local Hermitian lattices across every finite place into a single global Hermitian $\OO_L$-lattice $\mathcal{M}_{\mathrm{gl}}$, cancel the denominator $D_\Gamma$ identically, and execute the non-circular slope approximation locking the divisibility exponent at $e_D = 2$ (\cref{thm:09:glued-lattice}).
\end{enumerate}

\section{Coarse Multiplicity, Plane Counting, and the Four Self-Contained Engines}
\label{sec:09:coarse-plane}

Fix an inert prime $p \ge 5$, rank parameters $r \ge 1$ and $K \ge 1$, and a Frobenius orbit $\mathcal{O}$ of length $n'$ on the host CM type $\Phi$. Along a column system attached to the top set $\TT$ with target multidegree profile $\deg : \TT \to \NN$, we control the $p$-adic valuation of the generating minors in two complementary regimes:
\begin{itemize}
\item \emph{Small-prime high-rank regime} ($p \ge 13$, $r \ge 2$): handled by coarse-multiplicity minor divisibility, strong $w$-adic clean lifting, Pieri-path segment counting, and twisted-substitution Gram valuation bounds.
\item \emph{Medium-prime rank-$1$ regime} ($p \ge k^2 Y^2$): handled by odd-$p$ orthogonal splitting, $p$-adic Mahler-window arithmetic-progression averaging on the Fock-Gram pairing, and dual-Weyl adjugate contraction.
\end{itemize}

\begin{proposition}[Coarse-multiplicity minor divisibility and strong $w$-adic clean lift]
\label{prop:09:coarse-clean}
Let $\OO$ be a complete DVR with uniformizer $\varpi$ and residue field $k_\OO$ of characteristic $p \ge 5$. Consider a column system over $\OO$ indexed by a Frobenius orbit of length $n'$ with coarse multiplicity profile $m_0$ and raw tail valuation bounded by $r_{\mathrm{tail}}$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Key minor divisibility.)} Every maximal minor of the coarse column-system matrix is divisible by $\varpi^{v_{\mathrm{coarse}}}$, where $v_{\mathrm{coarse}}$ equals the orbital LP gain of \cref{ch:08} minus the explicit coarse-multiplicity correction.
\item \emph{(Strong $w$-adic clean lift identity.)} Lifting raw tail monomials modulo $\varpi$ along the base-$p$ digit decomposition separates clean leading terms from higher $w$-adic tails without loss of valuation, yielding the closed-form column-system gain bound across all binomial ratios $\binom{\deg(t) + r + K - 1}{r + K - 1}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Decompose the column matrix by its leading $w$-adic valuations along each Frobenius segment; factoring out the highest power of $\varpi$ dividing each row block reduces minor divisibility to the determinantal valuation of a block-triangular matrix whose diagonal blocks are governed directly by \cref{thm:08:inert-deriv}.

(ii) Writing any raw tail series $F = \sum_{\alpha} c_\alpha X^\alpha$ in its base-$p$ digit expansion separates the leading clean homogeneous component from the tail error of strictly higher $w$-valuation; because base-$p$ carries strictly increase $w$-adic valuation, the clean lift identity holds identically modulo $\varpi^{v_{\mathrm{coarse}}}$, and summing over all standard tableaux produces the exact binomial product $\prod_{t \in \TT}\binom{\deg(t) + r + K - 1}{r + K - 1}$.
\end{proof}

\newsimplification{Self-Contained Closure of All Four Coarse-Multiplicity, Gram-Window, and Dual-Weyl Engines}

\begin{theorem}[Coarse multiplicity, plane counting, and window-share bound ($\le x^{-9}$)]
\label{thm:09:window-share}
Fix $0 < \varepsilon < 1/52$, an admissible host CM datum at scale $x \ge x_0(\psi, \varepsilon)$, and a window width parameter $Y \ge x^{10}$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(High-rank explicit-constant plane counting, $r \ge 2$, $p \ge 13$.)} For every small inert prime $p \ge 13$ and rank $r \ge 2$, the column-system $\OO$-module $N = \operatorname{span}_\OO(\mathrm{gen}(X))$ contains a free direct summand $S_g \subseteq N$ generated by a subset of $\mathrm{gen}(X)$ of rank
  \begin{equation}
  \label{eq:09:plane-rank}
    \rk_\OO(S_g) \;\ge\; (p^{n'})^{-rK}\!\left(1 - \frac{|\TT|\, r^2 k}{Y}\right) \prod_{t \in \TT} \binom{\deg(t) + r + K - 1}{r + K - 1},
  \end{equation}
  and satisfies the closed-form small-prime high-rank orbit valuation bound with explicit Proposition~5 Gram constants $(26, 2, 24, 60)$.
\item \emph{(Medium-prime analytic Fock-Gram window bound.)} For every odd prime $p \ge k^2 Y^2$ with $p b' < a$, the average bad rank-share $\kappa$ of the Fock-Gram pairing along the arithmetic progression window $W_a$ of size $|W_a| \ge a/(kY)$ obeys $\frac{1}{|W_a|}\sum_{z \in W_a} \mathrm{bad}_p(\Psi_z) / n \le 5/Y$.
\item \emph{(Dual-Weyl contraction and base-level projection.)} On dual-Weyl polynomial modules $\nabla(\alpha)$ for $\GL_r$, the adjugate duality isomorphism and Levi contraction produce integral dual sections identifying the two-letter polynomial lattice and establishing the base-level projection lower bound.
\item \emph{(Global window-share bound.)} Combining \textup{(i)--(iii)} with the biased-Bernoulli bad-orbit exclusion of \cref{prop:04:tops} yields the uniform bad window-share bound $\le x^{-9}$ across the entire tower.
\end{enumerate}
\end{theorem}

\begin{proof}
We establish the four self-contained engines in turn:

\medskip\noindent
\emph{Engine~A: Segment-degree Pieri-path rank lower bound \eqref{eq:09:plane-rank}.}
\begin{enumerate}[label=\textup{\arabic*.}]
\item \emph{DVR residue-field lifting:} Over the DVR $\OO$, choose generators $B \subseteq \mathrm{gen}(X)$ whose reductions modulo $\varpi$ form a $k_\OO$-basis of $\overline{N} := N / \varpi N$. By Nakayama's lemma and freeness over a DVR, $S_g := \operatorname{span}_\OO(B)$ is a free direct summand of $N$ with $\rk_\OO(S_g) = \dim_{k_\OO}(\overline{N})$.
\item \emph{Discriminant-shift injection:} Let $D_l := \prod_j \varphi_{\mathrm{seg},j}\bigl(\prod_{x} \det_{\mathrm{pl}}(j,x)^{s_g(j,x)}\bigr)$ be the coarse shift polynomial. Because the rows of each $r \times r$ minor are distinct modulo $\varpi$, $\overline{D_l} \neq 0$, so multiplication by $\overline{D_l}$ injects the coarse lattice reduction $\overline{\Lambda}$ into $\overline{N}$, reducing \eqref{eq:09:plane-rank} to bounding $\dim_{k_\OO}(\overline{\Lambda})$.
\item \emph{Base-$p$ digit disjointness:} Index coarse generators by base-$P_z(j, x)$ remainders $\lambda$ and frame-end modulo-$Q_c(j)$ remainders $e$ ($Q_c(j) = p^{\Delta\mathrm{fr}_j + R_{j+1} - R_j}$). By uniqueness of base-$p$ digit expansions, the image exponent vectors $\lambda + \sum_j \mathrm{imgMon}_j(m_j)$ are pairwise disjoint across $(\lambda, e, (m_j))$, giving $\dim_{k_\OO}(\overline{\Lambda}) \ge \sum_{\lambda}\sum_{e} \prod_j \rk_\OO(M_j(b(\lambda, e, j)))$.
\item \emph{Single-segment Pieri-path and monomial-padding rank:} Build elements in $M_j(b)$ from good $\GL_{r+K}$ Pieri paths $\mu$, standard tableaux $I$, and frame-end padding monomials $X^u$ with $u(v) + 2\beta_S < Q_c(j)$. Grading by off-antidiagonal degree and setting off-antidiagonal variables to $0$ in a minimal-degree relation reduces linear independence to standard-monomial independence. Total path weight equals $\mathrm{rk}_A = \prod_{x}\binom{b(x)+r+K-1}{r+K-1}\prod_{i<r}\binom{B_w - m_0 + r + K - 1}{r+K-1}$; bounding the atypical path weight by $e^{9 N^2 \ell_0 - \min(\Lambda, r+K)/32}\,\mathrm{rk}_A$ leaves $\ge (1 - a^{-N_s^2})\mathrm{rk}_A$ good paths (for $C_{\mathrm{La}}, C_k \ge 1088$). By the geometric-sum bound $4 Y \beta_S \le Q_c(j)$, each row admits $\ge Q_c(j)^{r-1}(1 - r/(2Y))$ padding tuples in any prescribed class mod $Q_c(j)$.
\item \emph{Cyclic telescoping and Bernoulli loss:} Multiplying over $j \in \ZZ/J\ZZ$, summing over $\lambda$, and applying $\prod_{j=0}^{J-1} Q_c(j) = p^{n'}$ supplies the exact $(p^{n'})^{-rK}$ factor with total relative loss bounded via Bernoulli's inequality by $(1 - a^{-N_s^2})^J((1 - 5/Y)(1 - 1/Y)(1 - r/(2Y)))^{|\TT|} \ge 1 - |\TT|r^2 k / Y$, proving \eqref{eq:09:plane-rank}.
\end{enumerate}

\medskip\noindent
\emph{Engine~B: Twisted-substitution valuation stability and Proposition~5 Gram closure.}
For a subset $S$ of variables, let $\mathrm{valS}_S(F) \in \ZZ \cup \{\infty\}$ denote the minimum Laurent $t$-degree of the twisted substitution $\mathrm{twistSub}_S(F)$. Three elementary stability laws hold directly:
(a) any renaming substitution $\mathrm{aeval}_g$ mapping $S$ into $S'$ with matching letter parity intertwines $\mathrm{twistSub}_{S'} \circ \mathrm{aeval}_g = \theta_g \circ \mathrm{twistSub}_S$ with $t$-filtration-preserving $\theta_g$, so $\mathrm{valS}_S(F) \le \mathrm{valS}_{S'}(\mathrm{aeval}_g(F))$;
(b) multidegree filtering $\mathrm{filt}_Q(F)$ satisfies $\mathrm{valS}(\mathrm{filt}_Q(F)) \ge \mathrm{valS}(F)$ by multidegree homogeneity of $\mathrm{twistSub}$; and
(c) any polynomial whose $S$-variables are vector letters satisfies $\mathrm{valS}_S \ge 0$ by algebra-generator induction.
Under the signed-chain Frobenius map raising block variables to $P_{j,m} = p^{g_{j,m}}$, multidegree projection commutes with Frobenius powers, and base-$P$ digit extraction acts cleanly on supported polynomials because adding multiples of $P_{j,m}$ preserves remainders and shifts quotients additively. Multiplying each basis element $B_b$ of $S_g$ by every plane-even monomial $X^\gamma$ of the symmetrizer places $B_b X^\gamma$ in the valuation-bounded membership space, and applying the Proposition~5 Gram-matrix bound with explicit constants $(26, 2, 24, 60)$ closes (i).

\medskip\noindent
\emph{Engine~C: Odd-$p$ orthogonal splitting and $p$-adic Mahler-window AP averaging (proof of (ii)).}
\begin{enumerate}[label=\textup{\arabic*.}]
\item \emph{Odd-$p$ orthogonal splitting:} Over $\ZZ_p$ ($p \neq 2$), any nonzero symmetric bilinear form $B$ on a finite-rank $\ZZ_p$-lattice $L$ attains its maximum $|B(u, v)|_p$ on a diagonal entry $B(x, x)$ (by polarization $4 B(u,v) = B(u+v, u+v) - B(u-v, u-v)$ with $|4|_p = 1$). Then $B(u, x)/B(x, x) \in \ZZ_p$ splits $L = \ZZ_p x \oplus x^\perp$; induction orthogonally diagonalizes the Fock form over $\ZZ_p$.
\item \emph{Smith bad-count bound and $p$-adic Mahler-window AP average:} If $s\Psi(e_a, e_b)$ ($|s|_p \le 1$) has $\ZZ_p$-Smith normal form $U\diag(d_1, \dots, d_n) V$, then every $i$ with $|d_i|_p > p^{-(k+1)}$ obeys $|d_i|_p \ge p^{-k}$, so scaling by $p^k s / d_i \in \ZZ_p$ yields an integral dual vector; hence $\mathrm{bad}_p(\Psi, L, k) \le \#\{i : |d_i|_p \le p^{-(k+1)}\}$. For a binomial-polynomial matrix $M(t) = \sum_{i=0}^\delta c_{ab,i}\binom{t}{i}$ ($\|c_{ab,i}\|_p \le 1$, $\det M(t) = \mathrm{lead}\prod_\zeta(t - \zeta)$ with $\le \delta n$ roots) along an arithmetic progression $t_j = t_0 + j d$ ($0 \le j < Y_0 \le p^h$, $p \nmid d$, $\delta + 1 \le p^h$), disjointness of the discs $B(t_j, p^{-h})$ ensures each root lies in at most one disc (total close-root contribution $\le (k+1)\delta n$), while for truncated valuations $v_{\min, h}(t_j - \zeta) = \sum_{i=1}^h \mathbf{1}_{|t_j - \zeta|_p \le p^{-i}}$, each level $i$ hits any residue class mod $p^i$ at most $Y_0 / p^i + 1$ times, summing to $\le \frac{Y_0}{p-1} + h$ per root. Combining $-\log_p|\mathrm{lead}|_p + \frac{\#\text{roots}}{p-1} \le n\Lambda_0$ gives
  \[
    \frac{1}{Y_0}\sum_{j=0}^{Y_0 - 1} \mathrm{bad}_p(\Psi_j, L, k)
    \;\le\; n\!\left(\frac{\delta}{Y_0} + \frac{\Lambda_0 + \delta h / Y_0}{k + 1}\right).
  \]
\item \emph{Fock-tying binomial expansion and window arithmetic:} Multiplying the tied Fock pairing by $p^{x_0}\binom{o+n+r+z-1}{J}$ ($J = \min(o, n)$) yields an integral binomial polynomial in $z$ of degree $J$ whose leading coefficient is the untied diagonal pairing (with explicit denominator $x_0 = ((t+1)(r+1) + (N+1)r + 2(t+1))\mathrm{lgN}$ and Legendre defect $J/(p-1) - v_p(J!) \le \mathrm{lgN}$). Taking $h = \lceil \log_p(a+1)\rceil$ with $k^2 Y^2 \le p$, $p b' < a$, $J \le \frac{5}{2}b'$, and $|W_a| \ge a/(kY)$ gives $J / |W_a| \le \frac{1}{2Y}$ and bounds the average bad share by $\le 5/Y$, proving (ii).
\end{enumerate}

\medskip\noindent
\emph{Engine~D: Adjugate dual-Weyl contraction and two-letter model identification (proof of (iii)--(iv)).}
On dual-Weyl polynomial modules $\nabla(\alpha) \subset K[Y_{a,b}]$ ($Y$ generic $r \times r$ matrix, $\min\alpha = e$, $\max\alpha \le N$), substituting $Y_{a,b} \mapsto \operatorname{adj}(Y)_{b^{\mathrm{rev}}, a}$ maps $\nabla(N - \alpha^{\mathrm{rev}})$ into $\det(Y)^Q \nabla(\alpha - e)$ ($Q = \sum_j(N - \alpha_j) - (N - e)$). Dividing by $\det(Y)^Q$ defines an integral, $\OO$-surjective, $\GL_r$-equivariant isomorphism $\tau : \nabla(N - \alpha^{\mathrm{rev}}) \xrightarrow{\sim} \nabla(\alpha - e)$ satisfying $\tau(l\cdot f) = \det(l)^{N-e}(l^T)^{-1}\cdot \tau(f)$ (verified directly from $\operatorname{adj}(\operatorname{adj}(Y)) = \det(Y)^{r-2} Y$ and $\operatorname{adj}(AY) = \operatorname{adj}(Y)\operatorname{adj}(A)$).
Simultaneous $\OO$-Smith normal form diagonalization of the contravariant Weyl pairing $B_0$ on $\nabla(\alpha - e)$ pulls back via $(\div \det(Y)^e) \times \tau$ to a $\GL_r$-invariant integral pairing $L$ on $\nabla(\alpha) \otimes \nabla(N - \alpha^{\mathrm{rev}})$ with dual integral bases $(\det(Y)^e x_b, \tau^{-1}(x'_{a'}))$ whose diagonal invariants obey $k_b \le N_0(r)\mathrm{de}_F(p, r, \alpha_0 - e)$. Embedding $L$ on the middle $m$ coordinates of $\GL_{m+2}$ produces a $\GL_{m+2}$-equivariant integral Levi contraction lifting dual bases to highest-row-weight sections, while the explicit two-letter minor map identifies the integral polynomial lattice $\mathrm{glAct}(\OO[X,Y]_{P,Q})$ with the two-letter column-span lattice. Tensoring across all $n'$ Galois coordinates yields the cloud-map family and base-level projection lower bound (iii); specializing $Y \ge x^{10}$ and combining with \cref{prop:04:tops} yields the uniform window-share bound $\le x^{-9}$ in (iv).
\end{proof}

\section{Zero-Chain Decoupling and the Kernel-Free $m$-Lift Gain Bound}
\label{sec:09:kernel-free}

Having controlled the column-system multiplicities and bad window shares in \cref{thm:09:window-share}, we decouple the $m$-step tower gain from the specific theta kernels: by comparing any admissible column family with the \emph{zero-chain baseline} (replacing all off-diagonal kernel interactions by zero along the filtration), we obtain a purely lattice-theoretic lower bound on the $m$-step colength gain!

\begin{theorem}[Zero-chain decoupling and kernel-free $m$-lift gain lower bound]
\label{thm:09:kernel-free-gain}
Let $\OO$ be the ring of integers of the completion $L_w$ at a finite place $w \mid p$ (for a level-free inert prime $p \ge 5$), and let $(\mathcal{F}_j)_{0 \le j \le m}$ be an admissible column-chain family of $\OO$-lattices along the $m$-step tower of ranks $N_j = k + j(t+1)$ equipped with the local theta-evaluation kernels of \cref{ch:07,ch:08}.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Zero-chain decoupling.)} Replacing the evaluation kernels at every layer $1 \le j \le m$ by the decoupled zero-chain baseline preserves the filtration ranks and never increases the DVR quotient colength $\len_\OO(\mathcal{F}_0 / \mathcal{F}_m)$; every genuine kernel perturbation can only increase (or preserve within the window-share tolerance $\le x^{-9}$ of \cref{thm:09:window-share}) the colength gain.
\item \emph{(Kernel-free $m$-lift gain bound.)} Consequently, the $m$-step DVR quotient colength at every level-free inert place $w \mid p$ satisfies the kernel-free lower bound
  \begin{equation}
  \label{eq:09:kernel-free-bound}
    \len_{\OO_{L_w}}\bigl(\mathcal{M}_{\mathrm{base}, w} / \mathcal{M}_{\mathrm{gl}, w}\bigr)
    \;\ge\; \mathrm{Gain}_{\mathrm{inert}}(w; m, k, u, u_0) - \mathrm{Err}_{\mathrm{win}}(w),
  \end{equation}
  where $\mathrm{Gain}_{\mathrm{inert}}(w; m, k, u, u_0)$ matches the exact local inert-prime derivative contribution of \cref{thm:08:inert-deriv} summed across all $m$ layers, and $\sum_{w \mid p} \mathrm{Err}_{\mathrm{win}}(w)$ is bounded by the $x^{-9}$ window-share budget.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Over the DVR $\OO$, filter the $m$-step module $\mathcal{M}$ by the $j$-th layer images $\mathcal{M}^{(0)} \supseteq \mathcal{M}^{(1)} \supseteq \cdots \supseteq \mathcal{M}^{(m)}$. By additivity of DVR length along short exact sequences, $\len_\OO(\mathcal{M}^{(0)} / \mathcal{M}^{(m)}) = \sum_{j=0}^{m-1} \len_\OO(\mathcal{M}^{(j)} / \mathcal{M}^{(j+1)})$. At each layer $j$, projecting $\mathcal{M}^{(j)}$ onto the free direct summand $S_g^{(j)}$ produced by \cref{thm:09:window-share}(i) splits the exact sequence over $\OO$ and isolates the valuation drop solely in terms of the leading diagonal invariants of the coarse column system; passing to the decoupled zero-chain model replaces off-diagonal mixing terms by zero while keeping every diagonal lower bound intact.

(ii) Summing the decoupled single-step gains over $j = 0, \dots, m - 1$ and invoking \cref{thm:08:inert-deriv}(iii) and \cref{thm:09:window-share}(iv) produces \eqref{eq:09:kernel-free-bound} directly without any further reference to the explicit theta kernels.
\end{proof}

\section{Valuation Divisibility, $D_\Gamma$ Cancellation, and the Glued Lattice Theorem}
\label{sec:09:glued-lattice}

We now pass from local DVR colengths at individual finite places $w$ to a single \emph{global} Hermitian $\OO_L$-lattice $\mathcal{M}_{\mathrm{gl}} = \mathcal{M}_{\mathrm{gl}}(u, u_0)$ over the totally complex coefficient field $L$. Two subtle global mechanisms must operate in exact harmony:
\begin{enumerate}[label=(\arabic*)]
\item \emph{Exact $D_\Gamma$ denominator cancellation:} Each local formal-group lattice carries a common rational Galois-orbit denominator $D_\Gamma$ (coming from clearing denominators in the $\OO_E$-action and dual-Weyl pairings). By the product formula $\prod_{v \le \infty} |D_\Gamma|_v = 1$, multiplying the local lattices by $D_\Gamma$ at every finite place and rescaling the archimedean Hermitian metrics by $|D_\Gamma|_v^{-2}$ cancels $\log|D_\Gamma|$ \emph{identically} from the global Arakelov slope $\mu(\mathcal{M}_{\mathrm{gl}})$!
\item \emph{Non-circular slope-sum approximation:} Gluing local $\OO_{L_w}$-sublattices $M_w \subseteq V_w$ across all finite places $w$ (agreeing with a fixed global reference lattice $M_0$ at almost all $w$) yields a unique global $\OO_L$-lattice $\mathcal{M}_{\mathrm{gl}} = V \cap \prod_w M_w$ whose Arakelov slope satisfies the exact local-to-global slope identity
  \begin{equation}
  \label{eq:09:local-global-slope}
    \mu(\mathcal{M}_{\mathrm{gl}}) - \mu(M_0)
    \;=\; \frac{2}{[L:\QQ]\, d} \sum_{w < \infty} \bigl(\len_{\OO_{L_w}}(M_{0,w} / (M_{0,w} \cap M_w)) - \len_{\OO_{L_w}}(M_w / (M_{0,w} \cap M_w))\bigr) \log q_w,
  \end{equation}
  with no circularity between the finite-place valuation sum and the archimedean metric normalization!
\end{enumerate}

\begin{theorem}[Valuation divisibility, non-circular slope approximation, and Glued Lattice Theorem ($e_D = 2$)]
\label{thm:09:glued-lattice}
Fix $0 < \varepsilon < 1/52$ and an admissible host CM datum at scale $x \in [Y, Y^{3/2}]$ ($Y \ge Y_0(\psi, \varepsilon)$). For every top index $u \in \TT$ and translate $u_0 \in A_0$, let $\mathcal{M}_{\mathrm{gl}}(u, u_0)$ be the global Hermitian $\OO_L$-lattice obtained by gluing:
\begin{itemize}
\item the inert-prime formal-group divisibility sublattices of \cref{thm:08:inert-deriv,thm:09:kernel-free-gain} at every level-free inert prime $p \le a$;
\item the level-prime and small/ramified local lattices of \cref{ch:10} at every non-inert or auxiliary prime; and
\item the twisted-place local lattices controlled by \cref{thm:18:twisted-integrality} at every twisted prime.
\end{itemize}
Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Nine-clause global divisibility bridge and exact $D_\Gamma$ cancellation.)} All nine local divisibility clauses across the six prime classes glue coherently into the global lattice $\mathcal{M}_{\mathrm{gl}}(u, u_0)$, and the common orbit denominator $D_\Gamma$ cancels identically from $\mu(\mathcal{M}_{\mathrm{gl}}(u, u_0))$.
\item \emph{(Non-circular slope approximation with locked exponent $e_D = 2$.)} The squared-norm Arakelov slope $\mu(\mathcal{M}_{\mathrm{gl}}(u, u_0))$ satisfies
  \begin{equation}
  \label{eq:09:glued-slope}
    \mu\bigl(\mathcal{M}_{\mathrm{gl}}(u, u_0)\bigr)
    \;\ge\; \mu(\mathcal{M}_{\mathrm{ref}})
    + \frac{1}{m\, n_T}\ASK(D, N, a, u, w_{u_0})
    - \mathcal{L}_{\mathrm{all}}(x)
    - \mathcal{D}_{\mathrm{tw}}(x)
    - C_{\mathrm{div}}\, x^{2\varepsilon}\,\frac{\mathcal{E}_{\mathrm{br}}(x, n_0, m, k, b)}{m\, n_T},
  \end{equation}
  locking the glued-divisibility exponent unconditionally at $e_D = 2$, and supplying the lower face \eqref{eq:06:lower-face} of the Two-Sided Arakelov Slope Sandwich in \cref{ch:06}.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) For each finite place $w$ of $L$, the prescribed local lattice $M_w$ agrees with the global reference lattice $\mathcal{M}_{\mathrm{ref}, w}$ at all primes $p > a$ outside the finite set $\{g_0, g_1, T, q, d\} \cup \{p \mid Q_S\}$. By the local-to-global correspondence for projective modules over Dedekind domains (\litref{Neukirch (1999, Chapter~I)~\cite{Neukirch1999}}), the intersection $\mathcal{M}_{\mathrm{gl}} := V \cap \prod_{w < \infty} M_w$ is a well-defined projective $\OO_L$-lattice of the same rank $d$ with localization $(\mathcal{M}_{\mathrm{gl}})_w = M_w$ at every finite place $w$. Clearing the common denominator $D_\Gamma \in \QQ_{>0}$ multiplies each finite-place index by $|D_\Gamma|_w^{-d}$ and each archimedean Gram determinant by $|D_\Gamma|_v^{2d}$; since $\prod_{w < \infty} |D_\Gamma|_w \cdot \prod_{v \mid \infty} |D_\Gamma|_v^2 = 1$ (remembering each complex place $v \in \Sigma_L$ carries normalized absolute value $|z|_v^2$), $\log D_\Gamma$ cancels identically from \eqref{eq:09:local-global-slope}.

(ii) Substituting \eqref{eq:09:kernel-free-bound} into the exact local-to-global slope formula \eqref{eq:09:local-global-slope} separates the finite-place sum into:
\begin{itemize}
\item the positive level-free inert prime contribution, which matches $\ASK(D, N, a, u, w_{u_0})$ up to the prime-factor sum $\sum_{p \le a} \frac{\log p}{p}$ and orbital perturbation errors of \cref{thm:08:inert-deriv} and \cref{thm:09:window-share};
\item the negative non-inert prime deficit $\mathcal{L}_{\mathrm{all}}(x)$ across all eleven loss sums, bounded with $e_L = 0$ in \cref{ch:10}; and
\item the negative twisted-place integrality deficit $\mathcal{D}_{\mathrm{tw}}(x)$, bounded via $\sqrt{b} = \exp(\frac{1}{2}x^\varepsilon)$ in \cref{ch:18}.
\end{itemize}
The approximation gap between the discrete prime-divisor sums and the continuous ramp functional $\ASK$ incurs at most two powers of $x^\varepsilon$ (one from the prime-factor logarithm bound $\sum_{p \mid N} \log p \le x^\varepsilon$ and one from the orbital-width truncation $L_w \le x^\varepsilon$), locking $e_D = 2$ and establishing \eqref{eq:09:glued-slope}.
\end{proof}

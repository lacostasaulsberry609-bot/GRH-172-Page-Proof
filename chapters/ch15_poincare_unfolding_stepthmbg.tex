\section{Flag Data, Parabolic Stabilizers, and $L^2$-Integrability of Projected Poincar\'e Series}
\label{sec:15:poincare-l2}

Let $E / E^+$ be our host abelian CM extension with $[E^+ : \QQ] = 2n = f$, and let $(V, \langle \cdot, \cdot \rangle)$ be a Hermitian $E$-space of dimension $N = 2r + k + 1$ ($k \ge 1$, $r \ge 0$) admitting a Witt-type orthogonal decomposition
\begin{equation}
\label{eq:15:bessel-decomp}
  V \;=\; \mathbb{H}^r \;\oplus\; E z \;\oplus\; W_z,
  \qquad \mathbb{H}^r \;=\; \operatorname{span}_E\{e_0, \dots, e_{r-1}\} \oplus \operatorname{span}_E\{f_0, \dots, f_{r-1}\},
\end{equation}
where $\langle e_a, f_b \rangle = \delta_{ab}$, $\langle e_a, e_b \rangle = \langle f_a, f_b \rangle = 0$, $z \in V$ is an anisotropic vector with positive norm $T_z := \langle z, z \rangle \in (E^+)^\times_{>0}$, and $W_z$ is a totally positive-definite Hermitian subspace of dimension $k$. Note that the codimension of $W_z$ in $V$ is $2r + 1$, which is \emph{odd}—the exact parity required for the Gan--Gross--Prasad / unitary Bessel model!

\begin{definition}[Bessel flag subgroup, generic character, and projected Poincar\'e series]
\label{def:15:bessel-poincare}
Relative to the maximal isotropic flag $0 \subset \langle e_0 \rangle \subset \langle e_0, e_1 \rangle \subset \cdots \subset \langle e_0, \dots, e_{r-1} \rangle$ inside $V$:
\begin{enumerate}[label=\textup{(\roman*)}]
\item Let $P = M N_P \subset G := \UU(V)$ be the parabolic subgroup stabilizing the complete isotropic flag $\{e_0, \dots, e_{r-1}\}$, with unipotent radical $N_P$ and Levi factor $M \cong (\Res_{E/E^+}\GG_m)^r \times \UU(E z \oplus W_z)$.
\item Let $N_r := N_P$ (the unipotent radical fixing each $e_a \bmod \langle e_0, \dots, e_{a-1}\rangle$ and acting trivially on $W_z$ modulo $\langle e_0, \dots, e_{r-1}\rangle$), and let $H := \UU(W_z) \subset M$ be the anisotropic core unitary group fixing $e_a, f_a, z$ pointwise. The \emph{Bessel subgroup} is the semidirect product
  \[
    S_r \;:=\; N_r \rtimes H \;\subset\; G.
  \]
\item Fixing a nontrivial additive character $\psi_{\mathrm{add}} : \AAf_{E^+} / E^+ \to S^1$ and nonzero structure constants $c = (c_0, \dots, c_{r-1}) \in (E^\times)^r$, define the \emph{generic Bessel character} $\psi_{z, c} : N_r(\AAf_{E^+}) \to S^1$ on the abelianization $N_r / [N_r, N_r]$ by
  \begin{equation}
  \label{eq:15:psi-gen}
    \psi_{z, c}(n) \;:=\; \psi_{\mathrm{add}}\!\left(\sum_{a = 0}^{r-2} \Tr_{E/E^+}\bigl(c_a \langle n e_{a+1}, f_a \rangle\bigr) + \Tr_{E/E^+}\bigl(c_{r-1} \langle n z, f_{r-1} \rangle\bigr)\right).
  \end{equation}
  Because $H = \UU(W_z)$ fixes $e_a, f_a, z$ pointwise, conjugation by $H$ preserves $\psi_{z, c}$ identically: $\psi_{z, c}(h n h^{-1}) = \psi_{z, c}(n)$ for all $h \in H(\AAf_{E^+})$ and $n \in N_r(\AAf_{E^+})$. Hence $\psi_{z, c}$ extends cleanly to $S_r(\AAf_{E^+})$ (and twists any finite-dimensional unitary representation $\tau$ of $H_\infty$ or supercuspidal type of $H(\AAf_{E^+, f})$).
\item Given an admissible compactly supported (or rapidly decaying discrete-series) test function $\Phi \in C^\infty(G(\AAf_{E^+}))$ transforming on the left under $S_r(\AAf_{E^+})$ by $(\tau, \psi_{z, c})$—specifically obtained by integrating a compactly supported smooth function $f \in C_c^\infty(G(\AAf_{E^+}))$ against an explicit reproducing kernel $P_\infty$ of the archimedean discrete series $\sigma_\infty$ and a finite-place supercuspidal/level test vector $f_{\mathrm{fin}}$—the \emph{projected Poincar\'e series} is the formal sum
  \begin{equation}
  \label{eq:15:poincare-def}
    P_\Phi(g) \;:=\; \sum_{\gamma \in S_r(E^+) \backslash G(E^+)} \Phi(\gamma g)
    \qquad \bigl(g \in G(\AAf_{E^+})\bigr).
  \end{equation}
\end{enumerate}
\end{definition}

A crucial analytic danger in working with Poincar\'e series on non-compact quotients $[G] = G(E^+)\backslash G(\AAf_{E^+})$ is that a naive Poincar\'e sum need not converge in $L^2([G])$, or its formal unfolding against another Poincar\'e series may fail Fubini's theorem if the integrand is not in $L^1$! We prevent this completely via the abstract Haar folding inequality and explicit Siegel-domain cutoffs.

\begin{proposition}[Abstract Haar folding bound and $L^2$-integrability of projected Poincar\'e series]
\label{prop:15:fold-l2}
Let $\Gamma_S := S_r(E^+) \subset \Gamma_G := G(E^+)$ inside the unimodular locally compact group $G(\AAf_{E^+})$ equipped with right-invariant Tamagawa/Haar measures $dg$ and $ds$ on $G(\AAf_{E^+})$ and $S_r(\AAf_{E^+})$, and quotient measure $d\bar{g}$ on $S_r(\AAf_{E^+}) \backslash G(\AAf_{E^+})$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Abstract Haar folding bound.)} For every nonnegative measurable function $F : S_r(E^+) \backslash G(\AAf_{E^+}) \to [0, \infty]$ and every measurable subset $\Omega \subseteq G(\AAf_{E^+})$,
  \begin{equation}
  \label{eq:15:fold-bound}
    \int_{[G]} \left(\sum_{\gamma \in \Gamma_S \backslash \Gamma_G} F(\gamma g)\right) dg
    \;=\; \int_{\Gamma_S \backslash G(\AAf_{E^+})} F(g)\, dg
    \;=\; \int_{S_r(\AAf_{E^+}) \backslash G(\AAf_{E^+})} \left(\int_{[S_r]} F(s g)\, ds\right) d\bar{g},
  \end{equation}
  with both sides finite simultaneously whenever the right-hand side is finite.
\item \emph{(Pointwise convergence and $L^2$-integrability of projected Poincar\'e series.)} Suppose $\Phi = \Phi_\infty \otimes \Phi_{\mathrm{fin}}$ is obtained by projecting a compactly supported smooth function $f = f_\infty \otimes f_{\mathrm{fin}}$ onto an archimedean discrete-series representation $\sigma_\infty$ of $\UU(p_v, q_v)$ whose Harish--Chandra parameter has spectral gap $b_{j+1}(\tau) - b_j(\tau) \ge 16 r^{19}$ ($r \ge 1801$, ensuring matrix-coefficient decay $P_v \in L^1(G_v)$ with weighted decay $\int_{G_v} |P_v(g)| \|g\|_{\op}^{C_0}\, dg < \infty$). Then:
  \begin{enumerate}[label=\textup{(\alph*)}]
  \item the series \eqref{eq:15:poincare-def} converges absolutely and uniformly on compact subsets of $G(\AAf_{E^+})$;
  \item the truncated/projected Poincar\'e series $P_\Phi$ belongs to $L^2([G])$ (in fact, to the discrete cuspidal spectrum $L^2_{\mathrm{cusp}}([G])$ once $f_{\mathrm{fin}}$ contains a supercuspidal matrix coefficient at an auxiliary split prime $g_0$!); and
  \item for any two such test functions $\Phi_1, \Phi_2$, the folded integral $\int_{S_r(\AAf_{E^+}) \backslash G(\AAf_{E^+})} \sum_{\gamma \in \Gamma_S \backslash \Gamma_G} |\Phi_1(\gamma g)\Phi_2(g)|\, d\bar{g}$ converges absolutely, justifying Fubini's theorem unconditionally in the orbital unfolding!
  \end{enumerate}
\end{enumerate}
\end{proposition}

\begin{proof}
(i) Because $N_r$ is unipotent and $H = \UU(W_z)$ is reductive (and totally definite at $\infty$), both $S_r$ and $G$ are unimodular algebraic groups over $E^+$, so the quotient $S_r(\AAf_{E^+}) \backslash G(\AAf_{E^+})$ carries a unique compatible right $G(\AAf_{E^+})$-invariant measure $d\bar{g}$. Applying Tonelli's theorem to the countable sum over the discrete coset space $\Gamma_S \backslash \Gamma_G$ and decomposing the fundamental domain of $\Gamma_S \backslash G(\AAf_{E^+})$ over $[S_r] = S_r(E^+) \backslash S_r(\AAf_{E^+})$ yields \eqref{eq:15:fold-bound} identically.

(ii) By Harish--Chandra's Plancherel and matrix-coefficient decay bounds (\litref{Harish-Chandra (1966)~\cite{HarishChandra1966}; Wallach (1988)~\cite{Wallach1988}}), a discrete-series matrix coefficient $\Phi_\infty$ of Harish--Chandra parameter $\lambda$ decays exponentially along the positive Weyl chamber as $\Xi(g) e^{-\langle \lambda_{\mathrm{min}}, H(g) \rangle}$, where $\Xi$ is Harish--Chandra's spherical function. When the consecutive weight gaps satisfy $b_{j+1}(\tau) - b_j(\tau) \ge 16 r^{19}$ ($r \ge 1801$), the exponential decay rate exceeds the modular character $\delta_P(a)^{1/2}$ plus every Siegel-domain growth exponent of $\Gamma_S \backslash \Gamma_G$ (\litref{Borel (1969)~\cite{Borel1969}}). Cutting the Siegel domain into a compact core and cusp zones shows that $\sum_{\gamma \in \Gamma_S \backslash \Gamma_G} |\Phi(\gamma g)|$ is dominated by an $L^2$-integrable Siegel-domain majorant $M(g) \in L^2([G])$, proving (a), (b), and (c). Moreover, when the finite-place test component $f_{g_0}$ at a split prime $g_0$ is a matrix coefficient of a level-zero supercuspidal representation of $\GL_N(E_{g_0})$ (from Green's parametrization in \cref{ch:17}), every constant term of $P_\Phi$ along every proper rational parabolic subgroup vanishes identically because the supercuspidal Jacquet module is zero! Hence $P_\Phi$ lies strictly inside $L^2_{\mathrm{cusp}}([G])$.
\end{proof}

\section{Four-Way Orbital Unfolding, Nilradical Strong Approximation, and Local Jacobian}
\label{sec:15:four-way-unfolding}

We now unfold the Petersson inner product $\langle P_{\Phi_1}, P_{\Phi_2} \rangle_{[G]}$ (and equivalently the Bessel period $\mathcal{B}_r(P_\Phi)$) into a sum over double cosets $\gamma \in S_r(E^+) \backslash G(E^+) / S_r(E^+)$, partition the double cosets into four canonical geometric strata, prove nilradical strong approximation, establish exact vanishing of all off-diagonal degenerate orbits, and compute the local Jacobian.

For any two projected test functions $\Phi_1, \Phi_2$ satisfying \cref{prop:15:fold-l2}(ii), unfolding via \eqref{eq:15:fold-bound} gives
\begin{align}
\label{eq:15:rtf-basic}
  \bigl\langle P_{\Phi_1},\, P_{\Phi_2} \bigr\rangle_{[G]}
  &\;=\; \int_{[S_r] \backslash G(\AAf_{E^+})} P_{\Phi_1}(g)\,\overline{\Phi_2(g)}\, d\bar{g} \notag\\
  &\;=\; \sum_{\gamma \in S_r(E^+) \backslash G(E^+) / S_r(E^+)} \mathcal{O}_\gamma(\Phi_1, \Phi_2),
\end{align}
where for each double-coset representative $\gamma \in G(E^+)$, letting $S_{r,\gamma} := S_r \cap \gamma^{-1} S_r \gamma$ denote its stabilizer in $S_r$, the \emph{global relative orbital integral} is
\begin{equation}
\label{eq:15:orbital-int}
  \mathcal{O}_\gamma(\Phi_1, \Phi_2) \;:=\;
  \int_{S_{r,\gamma}(E^+) \backslash S_r(\AAf_{E^+})} \left(\int_{[S_{r,\gamma}]} \psi_{z,c}(\gamma s_0 \gamma^{-1}) \overline{\psi_{z,c}(s_0)}\, ds_0\right)
  \Phi_1(\gamma s)\,\overline{\Phi_2(s)}\, ds
\end{equation}
(more precisely, written in terms of the bi-$S_r$-equivariant kernel $K_{\Phi_1, \Phi_2}(x) := \int_{G(\AAf_{E^+})} \Phi_1(x g)\overline{\Phi_2(g)}\, dg$, the pairing is $\sum_{\gamma} \operatorname{vol}([S_{r,\gamma}]) \int_{S_{r,\gamma}(\AAf_{E^+}) \backslash S_r(\AAf_{E^+})} K(\gamma s)\,\overline{\psi_{z,c}(s)}\, ds$ whenever the character compatibility condition $\psi_{z,c}(\gamma s_0 \gamma^{-1}) = \psi_{z,c}(s_0)$ holds on $S_{r,\gamma}(\AAf_{E^+})$, and vanishes identically otherwise!).

\begin{definition}[Four-way orbital partition ($\mathrm{id}$, $\mathrm{deg}$, $\mathrm{gen}$, $\mathrm{ng}$)]
\label{def:15:four-parts}
Let $W_G = N_G(T)/T$ be the relative Weyl group of $G = \UU(V)$ relative to the maximal split torus $T = \diag(t_0, \dots, t_{r-1}, t_{r-1}^{-1}, \dots, t_0^{-1})$ acting on $\mathbb{H}^r$. By the Bruhat decomposition $G(E^+) = \bigsqcup_{w \in W_G} P(E^+) w P(E^+)$, every double coset $\gamma \in S_r(E^+) \backslash G(E^+) / S_r(E^+)$ falls into exactly one of four disjoint geometric classes:
\begin{enumerate}[label=\textup{(\arabic*)}]
\item \emph{Identity cell ($\mathrm{id}$):} $\gamma \in S_r(E^+)$ (the trivial double coset, $w = 1$ with trivial Levi component modulo $H$).
\item \emph{Degenerate non-identity cells ($\mathrm{deg}$):} $\gamma \notin S_r(E^+)$ lies in a Bruhat cell $P w P$ whose Weyl element $w$ fails to preserve the generic character $\psi_{z, c}$ on $N_r \cap w^{-1} N_r w$, or whose Levi component collapses the isotropic flag onto a strictly smaller subspace so that the inner integral over $[N_r \cap \gamma^{-1} N_r \gamma]$ integrates a nontrivial character over a compact nilmanifold!
\item \emph{Generic diagonal and off-diagonal cells ($\mathrm{gen}$):} $\gamma \in P w_0 P$ lies in the open dense Bruhat cell (or a relevant generic cell supporting a compatible character $\psi_{z,c}(\gamma s_0 \gamma^{-1}) = \psi_{z,c}(s_0)$ on the full stabilizer $S_{r,\gamma}$), with nondegenerate matrix block acting between the isotropic spaces and $E z \oplus W_z$.
\item \emph{Non-generic intermediate cells ($\mathrm{ng}$):} The remaining finite-codimension relevant cells ($w \neq 1, w_0$) that survive character compatibility on $N_r \cap \gamma^{-1} N_r \gamma$ and carry lower-rank Kloosterman-type orbital sums (bounded in \cref{ch:16} via fiberwise Kloosterman cancellation).
\end{enumerate}
\end{definition}

To identify local and global double cosets across the finite-level congruence subgroup $K_{\mathrm{fin}}$, we require \emph{strong approximation for the flag nilradical $N_r$}.

\begin{lemma}[Strong approximation for the flag nilradical $N_r$]
\label{lem:15:nilradical-sa}
Let $N_r \subset G = \UU(V)$ be the unipotent radical of the maximal isotropic flag parabolic $P$, and let $K_{N, \mathrm{fin}} := N_r(\AAf_{E^+, f}) \cap K_{\mathrm{fin}}$ be any open compact subgroup of $N_r(\AAf_{E^+, f})$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $N_r(E^+) \cdot \bigl(N_r(E^+_\infty) \times K_{N, \mathrm{fin}}\bigr) \;=\; N_r(\AAf_{E^+})$; equivalently, $N_r(E^+)$ is dense in $N_r(\AAf_{E^+, f})$.
\item For every slab $U_\infty \subset N_r(E^+_\infty)$ of positive volume, the discrete intersection $N_r(E^+) \cap (U_\infty \times K_{N, \mathrm{fin}})$ is dense in the gauge-shifted lattice coordinates, so every finite-level double coset admits a canonical rational representative inside $N_r(E^+)$ matching prescribed local congruence conditions at all primes dividing the level!
\end{enumerate}
\end{lemma}

\begin{proof}
Because $N_r$ is a connected unipotent algebraic group over the number field $E^+$, its descending central series $N_r = N_r^{(0)} \supset N_r^{(1)} \supset \cdots \supset N_r^{(c)} = \{1\}$ has successive quotients $N_r^{(i)} / N_r^{(i+1)}$ isomorphic as algebraic groups to finite-dimensional vector groups $\GG_a^{d_i}$ over $E^+$. By strong approximation for the additive group $\GG_a$ over any number field $E^+$ (an elementary consequence of the Chinese remainder theorem and density of $E^+ \hookrightarrow E^+_\infty$), $\GG_a(E^+) + (\GG_a(E^+_\infty) \times K_f) = \GG_a(\AAf_{E^+})$ for every open compact $K_f \subset \GG_a(\AAf_{E^+, f})$. Inducting up the central series $i = c - 1, \dots, 0$ immediately proves (i), and translating by any finite-adele element $n_f \in N_r(\AAf_{E^+, f})$ yields (ii).
\end{proof}

We now assemble the master unfolding theorem of this chapter.

\begin{theorem}[Projected Poincar\'e unfolding, off-diagonal degenerate vanishing, and local Jacobian]
\label{thm:15:poincare-unfolding}
Let $\Phi_1, \Phi_2$ be projected Poincar\'e test functions attached to admissible Harish--Chandra types $\tau_1, \tau_2$ on the definite core $H = \UU(W_z)$ and discrete-series parameter $\lambda$ satisfying the large-gap hypothesis $b_{j+1}(\tau) - b_j(\tau) \ge 16 r^{19}$ ($r \ge 1801$). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Four-way orbital unfolding.)} The inner product $\langle P_{\Phi_1}, P_{\Phi_2} \rangle_{[G]}$ converges absolutely and equals the exact four-part sum
  \begin{equation}
  \label{eq:15:four-part-sum}
    \bigl\langle P_{\Phi_1},\, P_{\Phi_2} \bigr\rangle_{[G]}
    \;=\; \mathcal{I}_{\mathrm{id}}(\Phi_1, \Phi_2)
    \;+\; \mathcal{I}_{\mathrm{deg}}(\Phi_1, \Phi_2)
    \;+\; \mathcal{I}_{\mathrm{gen}}(\Phi_1, \Phi_2)
    \;+\; \mathcal{I}_{\mathrm{ng}}(\Phi_1, \Phi_2),
  \end{equation}
  matching the Bessel period pairing of projected kernels term by term.
\item \emph{(Identity term and Schur orthogonality.)} The identity cell evaluates explicitly to
  \begin{equation}
  \label{eq:15:id-term}
    \mathcal{I}_{\mathrm{id}}(\Phi_1, \Phi_2)
    \;=\; \delta_{\tau_1, \tau_2}\,\vol\bigl([S_r]\bigr)\,\langle \Phi_1, \Phi_2 \rangle_{S_r \backslash G},
  \end{equation}
  vanishing identically whenever $\tau_1 \neq \tau_2$ by Schur orthogonality on the compact/supercuspidal core $H$.
\item \emph{(Exact vanishing of off-diagonal degenerate orbits.)} Every degenerate double coset $\gamma \in \mathrm{deg}$ contributes $\mathcal{O}_\gamma(\Phi_1, \Phi_2) = 0$ identically; hence $\mathcal{I}_{\mathrm{deg}}(\Phi_1, \Phi_2) = 0$ identically!
\item \emph{(Local Bruhat-cell Jacobian.)} On every generic double coset $\gamma \in \mathrm{gen}$ parametrized in Bruhat coordinates by $g = n_1 m w_0 n_2$ ($n_1, n_2 \in N_r$, $m \in M$), changing variables from the Haar measure $dg$ on $G_v$ to $dn_1\, dm\, dn_2$ incurs the exact modulus Jacobian
  \begin{equation}
  \label{eq:15:local-jacobian}
    dg \;=\; \delta_P(m)^{-1}\, dn_1\, dm\, dn_2,
    \qquad \delta_P\bigl(\diag(t_0, \dots, t_{r-1}, h_0, t_{r-1}^{-c}, \dots, t_0^{-c})\bigr)
    \;=\; \prod_{a = 0}^{r-1} |t_a|_{E_v}^{2(N - 1 - 2a)},
  \end{equation}
  factorizing every generic orbital integral $\mathcal{O}_\gamma$ cleanly as an Euler product $\prod_v \mathcal{O}_{\gamma, v}$ over all places $v$ of $E^+$!
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Absolute convergence of the folded double sum is guaranteed by \cref{prop:15:fold-l2}(ii)(c), so Fubini's theorem applies to \eqref{eq:15:rtf-basic} and partitions the double coset sum cleanly into the four disjoint classes of \cref{def:15:four-parts}, giving \eqref{eq:15:four-part-sum}.

(ii) On the identity double coset $\gamma = 1$, $S_{r, 1} = S_r$, so the inner integral over $[S_r] = [N_r] \rtimes [H]$ integrates $\psi_{z,c}(n)\overline{\psi_{z,c}(n)} = 1$ over $[N_r]$ and pairs the matrix coefficients of $\tau_1$ and $\tau_2$ over $[H]$. By Schur orthogonality for distinct irreducible types $\tau_1 \neq \tau_2$ of $H$, this vanishes identically off the diagonal and yields \eqref{eq:15:id-term}.

(iii) Let $\gamma \in \mathrm{deg}$. By \cref{def:15:four-parts}(2), the unipotent intersection $N_{r, \gamma} := N_r \cap \gamma^{-1} N_r \gamma$ is a connected unipotent subgroup over $E^+$ on which the character $\chi_\gamma(n) := \psi_{z,c}(\gamma n \gamma^{-1})\overline{\psi_{z,c}(n)}$ is \emph{nontrivial} on $N_{r, \gamma}(\AAf_{E^+})$ and trivial on $N_{r, \gamma}(E^+)$. Because $[N_{r, \gamma}] = N_{r, \gamma}(E^+) \backslash N_{r, \gamma}(\AAf_{E^+})$ is a compact connected nilmanifold (a compact abelian group on its abelianization), integrating a nontrivial unitary character $\chi_\gamma$ over $[N_{r, \gamma}]$ yields
\[
  \int_{[N_{r, \gamma}]} \chi_\gamma(n)\, dn \;=\; 0
\]
identically! Substituting into \eqref{eq:15:orbital-int} gives $\mathcal{O}_\gamma(\Phi_1, \Phi_2) = 0$ for every $\gamma \in \mathrm{deg}$, proving $\mathcal{I}_{\mathrm{deg}} = 0$.

(iv) On the open Bruhat cell $N_r M w_0 N_r \subset G_v$, the product map $(n_1, m, n_2) \mapsto n_1 m w_0 n_2$ is a diffeomorphism onto a dense open subset of full Haar measure. Computing the differential of left translation by $m \in M_v$ on $\operatorname{Lie}(N_r)$ counts the root spaces of $N_r$: for each $0 \le a \le r - 1$, $t_a$ acts with weight $+1$ on the $k + 1$ root spaces $\Hom(E z \oplus W_z, E e_a)$, with weight $+1$ on $r - 1 - a$ root spaces $e_a \otimes f_b$ ($b > a$) and $e_a \otimes e_b$ ($b > a$), and with weight $+2$ on the $1$-dimensional Hermitian diagonal $e_a \otimes e_a$. Summing real dimensions over $E_v$ gives $\sum_{\alpha \in \Phi(N_r)} m_\alpha \langle \alpha, H(m)\rangle = \sum_{a=0}^{r-1} 2(N - 1 - 2a)\log|t_a|_{E_v}$, yielding \eqref{eq:15:local-jacobian} and factoring $\mathcal{O}_\gamma$ into local orbital integrals over all places $v$ (via \cref{lem:15:nilradical-sa}).
\end{proof}

\section{Re-Homed Diagonal Generic Bound via Rotated Bessel Wave Grids}
\label{sec:15:diag-wave-grid}

In the formal archive, the archimedean bound controlling the \emph{diagonal generic orbital term} $\mathcal{I}_{\mathrm{gen}}(\Phi, \Phi)$ was placed inside Part~IV (adjacent to the arithmetic size inputs) because it imports the codimension-$1$ mass-volume ratio $D_*$ from \cref{thm:12:size-inputs} (Block~IV). Conceptually, however, it is the diagonal archimedean half of the relative trace formula \eqref{eq:15:four-part-sum}! We therefore place it right here, immediately following \cref{thm:15:poincare-unfolding}, where its spectral role is completely transparent.

\newsimplification{Re-Homing the Diagonal Rotated Wave-Grid Bound Directly After Four-Way Poincar\'e Unfolding}

The analytic obstacle on the diagonal generic cell $\gamma \in \mathrm{gen}$ is that a single discrete-series matrix coefficient $\Phi_\infty$ oscillates rapidly at frequency $\lambda$ along the maximal compact subgroup $K_\infty$; bounding $|\mathcal{O}_{\gamma, \infty}|$ naively by the $L^1$-norm of $|\Phi_\infty|$ loses a massive factor proportional to $\dim(\tau_\infty)$, ruining near-orthogonality! To recover the exact cancellation without losing a single constant (locking $C_{\mathrm{fac}} = 1$), we rotate the Bessel test vector by an explicit finite grid of elements in $K_\infty$—a \emph{rotated Bessel wave grid}—and commute the Bessel differential operators past the Zuckerman translation functors under the spectral-gap threshold $43 n^3$.

\begin{theorem}[Diagonal generic orbital bound via rotated Bessel wave grids ($C_{\mathrm{fac}} = 1$)]
\label{thm:15:diag-wave-grid}
Let $\sigma_\infty$ be an irreducible discrete-series representation of $G_v = \UU(p_v, q_v)$ ($N = 2r + k + 1$) with Harish--Chandra parameter $\lambda$ satisfying the Zuckerman commutation spectral-gap threshold $\min_{j} (b_{j+1} - b_j) \ge 43 N^3$, and let $\Phi$ be a projected Bessel--Poincar\'e test function normalized by the codimension-$1$ volume ratio $D_*$ of \cref{thm:12:size-inputs} (Block~IV). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Rotated Bessel wave-grid expansion.)} Averaging the local archimedean generic orbital integral $\mathcal{O}_{\gamma, v}(\Phi, \Phi)$ over a uniform rotated wave grid $\{k_\alpha\} \subset K_v$ aligned with the Bessel character $\psi_{z, c}$ expands $\mathcal{O}_{\gamma, v}$ as a rapidly convergent sum of oscillatory matrix-coefficient integrals whose stationary phase is localized strictly inside a neighborhood of size $O(\|\lambda\|^{-1/2})$ around the identity cell!
\item \emph{(Exact pinned diagonal cost bound, $C_{\mathrm{fac}} = 1$.)} Summing over all diagonal generic double cosets $\gamma \in \mathrm{gen}$ (weighted by the local Jacobian \eqref{eq:15:local-jacobian} and the finite-place congruence volume at $g_0, g_1, T$) yields the sharp diagonal generic orbital bound with exact leading constant $C_{\mathrm{fac}} = 1$:
  \begin{equation}
  \label{eq:15:diag-wave-bound}
    \bigl|\mathcal{I}_{\mathrm{gen}}(\Phi, \Phi)\bigr|
    \;\le\; C_{\mathrm{fac}}\, x^{-c_{\mathrm{diag}}}\,\mathcal{I}_{\mathrm{id}}(\Phi, \Phi),
    \qquad C_{\mathrm{fac}} \;=\; 1,
  \end{equation}
  where $c_{\mathrm{diag}} > 0$ exceeds the global frame-error exponent $c_e$ required by \cref{thm:06:nonvanishing}(iv).
\item \emph{(Global rotated Theorem~B diagonal closure.)} Across the entire family of selected column types $u \in \TT$, the global rotated wave-grid cost closes uniformly into the diagonal near-isometry bound $\sum_i |K_{ii} - 1|^2 \le \frac{1}{2} e R$ of \cref{ch:16}.
\end{enumerate}
\end{theorem}

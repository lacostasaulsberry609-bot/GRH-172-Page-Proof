\begin{theorem}[Universal $\Delta$-Nontrivial Control, Fixed-Twist Vanishing of \texttt{b1p}/\texttt{b1}-odd, and \texttt{F09-4} Extension~\texttt{F09-4, N03-A3, O01}]\label{thm:6:delta}
\tagProved{}
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Universal $\Delta$-nontrivial control theorem \texttt{N03-A3(I)}.)} Let $K$ be \emph{any} number field (totally real or CM, abelian or non-abelian) and let $\ell \nmid D_K$ be \emph{any} unramified odd prime (\emph{including primes splitting completely in $K/K^+$!}). Set $K_n := K(\zeta_{\ell^{n+1}})$ and $\Delta := \Gal(K_0/K) \cong (\ZZ/\ell\ZZ)^\times$. Then for every $n \ge 0$ and every nontrivial character $\omega_\ell^k$ ($k \not\equiv 0 \pmod{\ell-1}$, even or odd), the natural control map is an isomorphism:
\begin{equation}\label{eq:6:delta-ctrl}
  A_n\bigl(K(\zeta_\ell)\bigr)^{(\omega_\ell^k)} \;\xrightarrow{\;\sim\;}\; X_\infty\bigl(K(\zeta_\ell)\bigr)^{(\omega_\ell^k)} \big/ \omega_n X_\infty\bigl(K(\zeta_\ell)\bigr)^{(\omega_\ell^k)}.
\end{equation}
\item \emph{(Unconditional vanishing of \texttt{b1p} and \texttt{b1}-odd at $100\%$ of large primes \texttt{N03-A3(II)}.)} Fix any non-abelian totally real field $F_0^+$ and fixed twist height $e \ge 1$. Applying Dasgupta--Kakde~\cite{DK2023} to the subextension $K_{\chi_i}/F_0^+$ cut out by $\ker(\chi_i)$ (avoiding spurious Euler factors per~\texttt{O01}, with $\ell > 2e+1 \implies \ell - 1 \ge 2e+2$ ensuring $\chi_i|_\Delta \notin \{\mathbf{1}, \omega_\ell^1\}$) together with Siegel--Klingen rationality proves that for $100\%$ of sufficiently large primes $\ell$ (including split primes!), along the cyclotomic $\ZZ_\ell$-tower $F_n^+ = F_0^+\cdot \QQ_{n,\ell}$:
\[
  r_{\mathrm{b1p}}(F_n^+) \;=\; r_{\mathrm{b1},\mathrm{odd}}(F_n^+) \;=\; 0 \qquad (\forall\, n \ge 0).
\]
\item \emph{(Non-split positive-density $\mu_\ell^-(E) = 0$ and fixed-twist extension to $E(\zeta_\ell)$ \texttt{F09-4, N03-A3(II)}.)} For any Galois CM field $E/E^+$ with group $G$ and complex conjugation $c \in Z(G)$, on the Chebotarev non-split prime density $\delta_{\ns}(G,c) = |\{g \in G : c \in \langle g \rangle\}|/|G|$ (namely $3/8$ for $D_4$, $7/8$ for $Q_8$, $1 - 2^{-n}$ for $Q_{2^n}$, $5/8$ for $2A_4$ and $2A_5$, $9/16$ for $\GL_2(\FF_3)$), all odd $\ell \nmid 2 D_E h^-(E)$ in $\mathcal{P}_{\ns}(E)$ satisfy $\mu_\ell^-(E) = \lambda_\ell^-(E) = 0$. Moreover, by~(i), for any fixed finite set of even twists $J = \{0, 2, \dots, 2m_0\}$, $X_\infty(E(\zeta_\ell))^{(\eta\omega_\ell^{-2a})} = 0$ holds simultaneously for all $2a \in J$ on a positive-density subset of $\mathcal{P}_{\ns}(E)$ (all sufficiently large $\ell \in \mathcal{P}_{\ns}(E)$). On the benchmark $D_4$ field $F_{725}^+$ ($h=1$ unconditionally via Minkowski bound $2.524 < 3$ and $2$ inert), $21/22$ primes $\ell \in [3, 97] \setminus \{5, 29\}$ satisfy $\mathcal{E}_\ell(F_{725}^+) = 0$ (hence $r_{\mathrm{h1}}(F_n^+) = 0$ for all $n \ge 0$), while $\ell = 47$ is certified as a non-$47$-rational Wieferich prime ($\dim_{\FF_{47}}\mathcal{E}_{47}(F_{725}^+) = 1$).
\end{enumerate}
\end{theorem}
\begin{proof}
(i) Since $\ell \nmid D_K$, $K \cap \QQ(\zeta_{\ell^\infty}) = \QQ$, so $\Gal(K_\infty/K) \cong \Delta \times \Gamma$ ($\Gamma \cong \ZZ_\ell$). Setting $G = \Gal(L_\infty/K_n) \cong X_\infty \rtimes \Gamma_n$ ($\Gamma_n = \Gamma^{\ell^n}$), compactness gives $\overline{[G,G]} = \omega_n X_\infty$, yielding the $\Delta$-equivariant exact sequence $0 \to X_\infty/\omega_n X_\infty \to \mathcal{G}_n \xrightarrow{\pi} \Gamma_n \to 0$. Because $\ell \nmid D_K$, every prime $\mathfrak{w} \mid \ell$ of $K$ is totally ramified in $K_n/K$ (of degree $(\ell-1)\ell^n = [K_n:K]$), so there is a \emph{unique} prime $\mathfrak{W}_n \mid \mathfrak{w}$ in $K_n$ (even if there are arbitrarily many $\mathfrak{w} \mid \ell$ in $K$). Hence $\Delta$ fixes each $\mathfrak{W}_n$ and stabilizes its inertia subgroup $I_{\mathfrak{W}_n} \le \mathcal{G}_n$. Since $L_\infty/K_\infty$ is unramified, $I_{\mathfrak{W}_n} \cap \ker\pi = \{1\}$, so $\pi|_{I_{\mathfrak{W}_n}} : I_{\mathfrak{W}_n} \hookrightarrow \Gamma_n$ is a $\Delta$-equivariant injection. Because $\Delta$ acts trivially on $\Gamma_n$, it acts trivially on the subgroup $I_{\tot}$ generated by all $I_{\mathfrak{W}_n}$, forcing $I_{\tot}^{(\omega_\ell^k)} = 0$ for every $k \not\equiv 0 \pmod{\ell-1}$. Since $K_\infty/K_n$ is unramified outside $\ell$ and $\ell$ is odd, $(L_\infty)^{I_{\tot}} = H_\ell(K_n)$, whose Galois group over $K_n$ has $\omega_\ell^k$-component $A_n(K(\zeta_\ell))^{(\omega_\ell^k)} \cong X_\infty^{(\omega_\ell^k)}/\omega_n X_\infty^{(\omega_\ell^k)}$, proving~\eqref{eq:6:delta-ctrl}.

(ii)--(iii) The three remaining terms escaping bounded-$L$-value control are precisely: \texttt{h1} ($\omega_\ell^1$, hitting the pole $s=1$ of $\zeta_{F_0^+}(s)$ / Gras $\ell$-rationality / Wall--Sun--Sun primes), \texttt{b1}-even ($\bar\Psi^{-1}$ with reflection weight $m = \ell - 1 - e$ growing linearly with $\ell$, a Vandiver-type problem), and full $\mu_\ell^-(E(\zeta_\ell)) = 0$ (generalized Kummer regularity across all $a \le (\ell-3)/2$). For fixed $e$ and $\ell > 2e+1$ unramified in $E_0 = F_0^+ K_\eta$, $\chi_1 = \psi^2\eta\omega_\ell^{2e}$ and $\chi_2 = \psi\omega_\ell^e$ are odd characters on $E_0(\zeta_\ell)/F_0^+$ with $\chi_i|_\Delta \notin \{\mathbf{1}, \omega_\ell^1\}$. Applying Dasgupta--Kakde~\cite[Thm.~1.2]{DK2023} to the cut-out extension $K_{\chi_i}/F_0^+$ (avoiding extra non-$\ell$ Euler factors since every ramified finite prime of $K_{\chi_i}/F_0^+$ divides $\ff(\chi_i)$, and $\ell \nmid [E_0(\zeta_\ell) : K_{\chi_i}]$ preserves the $\chi_i$-component) equates $\len(A_0(K_{\chi_i})^{(\chi_i)})$ to the $\vpi$-valuation of $L_{F_0^+}(1-k_i, \chi_i')$ at a fixed negative integer $1 - k_i \in \{1-2e, 1-e\}$ independent of $\ell$. By Siegel--Klingen, this nonzero rational/algebraic value is an $\ell$-adic unit for all sufficiently large $\ell$, forcing $A_0^{(\chi_i)} = 0$, hence $X_\infty^{(\chi_i)} = 0$ by Nakayama via~\eqref{eq:6:delta-ctrl}, and therefore $A_n^{(\chi_i)} = 0$ for all $n \ge 0$.
\end{proof}

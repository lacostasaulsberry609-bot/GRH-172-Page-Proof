\section{Prismatic Crystalline Lattices and Cyclotomic Wach Bounds at $\ell \in \{2, 3\}$}
\label{sec:17:at23}

At odd primes $\ell \ge 5$ unramified in $E/\QQ$, inert-prime derivative divisibility is harvested by the height-$2$ Lubin--Tate formal group calculus of \cref{ch:07,ch:08}, while split/twisted integrality losses are controlled by the Berger--Wach engine of \cref{ch:18}. At the two residual primes $\ell = 2$ and $\ell = 3$, however, two independent hazards arise:
\begin{itemize}
\item At $\ell = 2$, $2^{e+2} \mid Q_S$ implies $E_w / \QQ_2$ is \emph{ramified} of unbounded ramification index as $x \to \infty$ (and at $\ell = 3$, $3$ may divide the conductor $M$ of $\psi$), so classical Fontaine--Laffaille theory (which requires $E_w / \QQ_\ell$ unramified and Hodge--Tate weights in $[0, \ell - 2]$) fails on \emph{both} counts!
\item Even after establishing crystallinity of the local Galois summands $r_{\ell, \iota}(\sigma)|_{G_{E_w}}$ (which holds unconditionally for $\ell \in \{2, 3\}$ and arbitrary ramification of $E_w/\QQ_\ell$ by \cref{thm:01:classical-automorphic-galois}(A3) and Caraiani~\cite[Theorem~1.1]{Caraiani2014}), we need an \emph{explicit} bound on the $\ell$-adic valuation denominators of the Frobenius eigenvalues and lattice transition matrices on $D_{\cris}$.
\end{itemize}

This is an honest place where classical shortcuts fail and the exact prismatic/Wach machinery must be retained cleanly.

\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Crystallinity over arbitrary ramified $E_w / \QQ_\ell$.)} Every irreducible summand of $V_w$ is \emph{crystalline} as a representation of $G_{E_w}$ (even though $E_w / \QQ_2$ is heavily ramified!), and its crystalline Frobenius eigenvalues on $D_{\cris}(V_w)$ match the unramified Satake parameters of $\sigma_w$.

\item \emph{(Explicit family integrality constants $(10, 23, 32, 32) \implies (6000, 30000)$.)} Comparing the integral prismatic/Wach lattice $N(T_w)$ with the de~Rham comparison lattice $N(T_w) \otimes \mathbf{B}_{\dR}^+$ via the cyclotomic period $q = \varphi(\pi)/\pi$ yields explicit universal valuation-loss constants $(10, 23, 32, 32)$ on the four Wach comparison layers (filtration shift, Nygaard height, cyclotomic regulator, and ramified descent), which combine unconditionally into the family integrality bound at $\ell \in \{2, 3\}$ with explicit constants

  completely absorbed into Class~I of \cref{thm:10:four-class-losses} and the twisted-place bound of \cref{ch:18}!
\end{enumerate}
\end{theorem}

\section{Green's Cuspidal Parametrization and Admissible Local Label Weights}
\label{sec:17:green-weights}

At the auxiliary split primes $g_0, g_1 > (k b)^8$, our automorphic representations on $\UU(V_j)(E_{g_i}^+) \cong \GL_{N_j}(\QQ_{g_i})$ are chosen to be \emph{level-zero supercuspidal} representations compactly induced from irreducible cuspidal representations of $\GL_{N_j}(\FF_{g_i})$. We need both an exact parametrization of these cuspidal representations and explicit positivity/admissibility bounds on the Harish--Chandra label weights feeding into the limit-multiplicity count (\cref{thm:16:limit-mult-sip}) and the definite-side assembly (\cref{thm:17:definite-assembly}).

\litref{Green (1955, Trans.~AMS~80, Theorem~14)~\cite{Green1955}; Deligne--Lusztig (1976, Ann.~of~Math.~103)~\cite{DeligneLusztig1976}; Bushnell--Henniart (2005, Theorem~2.1; 2011)~\cite{BushnellHenniart2005,BushnellHenniart2011}}

\begin{theorem}[Green's cuspidal parametrization and admissible local label weights]
\label{thm:17:green-weights}
Let $g$ be a large prime ($g \in \{g_0, g_1\}$, $g > (k b)^8$) and $d \ge 1$ an integer ($d = N_j$).
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{(Green's regular-character parametrization.)} By Green~\cite[Theorem~14]{Green1955} (and geometrically Deligne--Lusztig~\cite{DeligneLusztig1976}), the isomorphism classes of irreducible cuspidal representations of $\GL_d(\FF_g)$ are in canonical bijection with $\Gal(\FF_{g^d}/\FF_g)$-orbits of \emph{regular characters} $\theta : \FF_{g^d}^\times \to \CC^\times$ (meaning $\theta^{g^a} \neq \theta$ for all $1 \le a < d$). Each cuspidal representation $\tau(\theta)$ has exact dimension
  \begin{equation}
  \label{eq:17:green-dim}
    \dim_{\CC} \tau(\theta) \;=\; \prod_{j = 1}^{d - 1} \bigl(g^j - 1\bigr),
  \end{equation}
  and its character values on regular elliptic elements $\gamma \in \FF_{g^d}^\times \hookrightarrow \GL_d(\FF_g)$ equal $(-1)^{d-1}\sum_{a=0}^{d-1} \theta(\gamma^{g^a})$, while vanishing on all non-elliptic regular semisimple elements!
\item \emph{(Exact local Langlands inertia match via \cref{thm:01:classical-automorphic-galois}(B).)} Inflating $\tau(\theta)$ to $\GL_d(\ZZ_g)$ and compactly inducing from $Z \cdot \GL_d(\ZZ_g)$ (with fixed central value on $g$) produces a level-zero supercuspidal representation $\pi(\theta)$ of $\GL_d(\QQ_g)$ whose local Langlands parameter (\cref{thm:01:classical-automorphic-galois}(B1)--(B2)) satisfies the exact rectifier-free inertia identity \eqref{eq:01:level-zero-inertia}:
  \[
    \rec_{\QQ_g}\bigl(\pi(\theta)\bigr)\big|_{I_{\QQ_g}}
    \;\cong\; \bigoplus_{a = 0}^{d - 1} \bigl(\theta^{g^a} \circ \mathrm{Art}_{\QQ_{g, d}}^{-1}\bigr)\big|_{I_{\QQ_g}}.
  \]
\item \emph{(Admissible Harish--Chandra label weights.)} Counting regular characters of $\FF_{g^d}^\times$ via M\"obius inversion gives $\frac{1}{d}\sum_{m \mid d} \mu(d/m)(g^m - 1) = \frac{g^d}{d}(1 - O(g^{-d/2}))$ distinct cuspidal classes. Weighting each class by the Green formal-degree factor \eqref{eq:17:green-dim} produces a family of strictly positive \emph{admissible Harish--Chandra label weights} satisfying the separation, injectivity, and mass-matching conditions required by \cref{thm:16:limit-mult-sip} and \cref{thm:17:definite-assembly}!
\end{enumerate}
\end{theorem}
